package org.apache.commons.lang3.math;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;
import static org.apache.commons.lang3.math.Fraction.*;

public class Fraction_GenTest {

    @Test
    public void testGetFractionDoubleException() {
        assertThrows(ArithmeticException.class, () -> getFraction(Integer.MAX_VALUE + 1.0));
    }

    @Test
    public void testGetFractionDoubleException_2() {
        assertThrows(ArithmeticException.class, () -> getFraction(Double.NaN));
    }

    @Test
    public void testGetFractionStringX_Y_ZFormat() {
        Fraction fraction = getFraction("1 2/3");
        assertEquals(5, fraction.getNumerator());
        assertEquals(3, fraction.getDenominator());
    }

    @Test
    public void testGetFractionStringDoubleFormat() {
        Fraction fraction = getFraction("1.5");
        assertEquals(3, fraction.getNumerator());
        assertEquals(2, fraction.getDenominator());
    }

    @Test
    public void testGetFractionStringY_ZFormat() {
        Fraction fraction = getFraction("2/3");
        assertEquals(2, fraction.getNumerator());
        assertEquals(3, fraction.getDenominator());
    }

    @Test
    public void testAddAndCheckOverflow() {
        assertThrows(ArithmeticException.class, () -> addAndCheck(Integer.MAX_VALUE, 1));
    }

    @Test
    public void testSubtractAndCheckUnderflow() {
        assertThrows(ArithmeticException.class, () -> subAndCheck(Integer.MIN_VALUE, 1));
    }

    @Test
    public void testMulAndCheckOverflow() {
        assertThrows(ArithmeticException.class, () -> mulAndCheck(Integer.MAX_VALUE, 2));
    }

    @Test
    public void testMulPosAndCheckNegative() {
        assertThrows(ArithmeticException.class, () -> mulPosAndCheck(Integer.MIN_VALUE, -1));

    public void testGetFractionDenominatorLarge_2() {

    }

    public void testGetFractionStringX_Y_ZFormat_2() {

        assertEquals(-5, fraction.getNumerator());
        assertEquals(3, fraction.getDenominator());
    }

    @Test
