package org.apache.commons.lang3.time;

import static org.junit.jupiter.api.Assertions.*;
import java.util.Calendar;
import java.util.Date;
import java.util.Iterator;
import java.util.NoSuchElementException;
import java.util.concurrent.TimeUnit;
import org.junit.jupiter.api.Test;

class DateUtils_GenTest {

    @Test
    void testDateIteratorNextThrowsNoSuchElementException() {
        Calendar start = Calendar.getInstance();
        Calendar end = (Calendar) start.clone();
        Iterator<Calendar> iterator = new DateUtils.DateIterator(start, end);
        assertThrows(NoSuchElementException.class, iterator::next);
    }

    @Test
    void testCeilingObjectDateBranches() {
        Date date = new Date();
        assertEquals(date, DateUtils.ceiling(date, Calendar.DATE));
        assertThrows(ClassCastException.class, () -> DateUtils.ceiling("not a date", Calendar.DATE));
    }

    @Test
    void testCeilingObjectCalendarBranches() {
        Calendar calendar = Calendar.getInstance();
        assertEquals(calendar.getTime(), DateUtils.ceiling(calendar, Calendar.DATE).getTime());
        assertThrows(ClassCastException.class, () -> DateUtils.ceiling("not a calendar", Calendar.DATE));
    }

    @Test
    void testGetFragmentInDaysCalendarBranches() {
        Calendar calendar = Calendar.getInstance();
        long fragmentInDays = DateUtils.getFragmentInDays(calendar, Calendar.DAY_OF_MONTH);
        assertEquals(fragmentInDays, DateUtils.getFragment(calendar, Calendar.DAY_OF_MONTH, TimeUnit.DAYS));
    }

    @Test
    void testGetFragmentInDaysDateBranches() {
        Date date = new Date();
        long fragmentInDays = DateUtils.getFragmentInDays(date, Calendar.DAY_OF_MONTH);
        assertEquals(fragmentInDays, DateUtils.getFragment(DateUtils.toCalendar(date), Calendar.DAY_OF_MONTH, TimeUnit.DAYS));
    }

    @Test
    void testIsSameDayPartialBranches() {
        Calendar cal1 = Calendar.getInstance();
        Calendar cal2 = (Calendar) cal1.clone();
        assertTrue(DateUtils.isSameDay(cal1, cal2));

        cal2.add(Calendar.YEAR, 1);
        assertFalse(DateUtils.isSameDay(cal1, cal2));
    }

    @Test
    void testIsSameInstantCalendarBranches() {
        Calendar cal1 = Calendar.getInstance();
        Calendar cal2 = (Calendar) cal1.clone();
        assertTrue(DateUtils.isSameInstant(cal1, cal2));

        cal2.add(Calendar.SECOND, 1);
        assertFalse(DateUtils.isSameInstant(cal1, cal2));
    }

    @Test
    void testIsSameLocalTimeBranches() {
        Calendar cal1 = Calendar.getInstance();
        Calendar cal2 = (Calendar) cal1.clone();
        assertTrue(DateUtils.isSameLocalTime(cal1, cal2));

        cal2.add(Calendar.SECOND, 1);
        assertFalse(DateUtils.isSameLocalTime(cal1, cal2));
    }

    @Test
    void testIteratorMonthSundayBranches() {
        Calendar calendar = Calendar.getInstance();
        Iterator<Calendar> iterator = DateUtils.iterator(calendar, DateUtils.RANGE_MONTH_SUNDAY);
        assertTrue(iterator.hasNext());
    }

    @Test
    void testIteratorWeekRelativeBranches() {
        Calendar calendar = Calendar.getInstance();
        Iterator<Calendar> iterator = DateUtils.iterator(calendar, DateUtils.RANGE_WEEK_RELATIVE);
        assertTrue(iterator.hasNext());
    }

    @Test
    void testIteratorWeekCenterBranches() {
        Calendar calendar = Calendar.getInstance();
        Iterator<Calendar> iterator = DateUtils.iterator(calendar, DateUtils.RANGE_WEEK_CENTER);
        assertTrue(iterator.hasNext());
    }

    @Test
    void testIteratorDefaultCaseThrowsIllegalArgumentException() {
        Calendar calendar = Calendar.getInstance();
        assertThrows(IllegalArgumentException.class, () -> DateUtils.iterator(calendar, 999));
    }

    @Test
    void testIteratorObjectDateBranches() {
        Date date = new Date();
        Iterator<?> iterator = DateUtils.iterator(date, DateUtils.RANGE_MONTH_SUNDAY);
        assertTrue(iterator.hasNext());
    }

    @Test
    void testIteratorObjectCalendarBranches() {
        Calendar calendar = Calendar.getInstance();
        Iterator<?> iterator = DateUtils.iterator(calendar, DateUtils.RANGE_MONTH_SUNDAY);
        assertTrue(iterator.hasNext());
    }

    @Test
    void testIteratorObjectThrowsClassCastException() {
        assertThrows(ClassCastException.class, () -> DateUtils.iterator("not a date or calendar", DateUtils.RANGE_MONTH_SUNDAY));
    }
}
