package org.apache.commons.lang3.time;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Calendar;
import java.util.TimeZone;

public class DurationFormatUtils_GenTest {

    @Test
    public void testTokenConstructorWithOptional() {
        Token token = new Token(new StringBuilder("test"), true, 1);
        assertEquals(1, token.optionalIndex);
    }

    @Test
    public void testTokenEqualsDifferentClass() {
        Token token1 = new Token(new StringBuilder("test"), false, -1);
        assertFalse(token1.equals("not a token"));
    }

    @Test
    public void testTokenEqualsDifferentValueClass() {
        Token token1 = new Token(new StringBuilder("test"), false, -1);
        Token token2 = new Token(123, false, -1);
        assertFalse(token1.equals(token2));
    }

    @Test
    public void testTokenEqualsDifferentCount() {
        Token token1 = new Token(new StringBuilder("test"), false, -1);
        Token token2 = new Token(new StringBuilder("test"), false, -1);
        token2.count = 2;
        assertFalse(token1.equals(token2));
    }

    @Test
    public void testTokenEqualsStringBuilder() {
        Token token1 = new Token(new StringBuilder("test"), false, -1);
        Token token2 = new Token(new StringBuilder("test"), false, -1);
        assertTrue(token1.equals(token2));
    }

    @Test
    public void testTokenEqualsNumber() {
        Token token1 = new Token(123, false, -1);
        Token token2 = new Token(123, false, -1);
        assertTrue(token1.equals(token2));
    }

    @Test
    public void testTokenEqualsSameReference() {
        Token token1 = new Token(new StringBuilder("test"), false, -1);
        assertTrue(token1.equals(token1));
    }

    @Test
    public void testTokenHashCode() {
        Token token = new Token(123, false, -1);
        assertEquals(Integer.valueOf(123).hashCode(), token.hashCode());
    }

    @Test
    public void testTokenToString() {
        Token token = new Token(new StringBuilder("test"), false, 3);
        assertEquals("testtesttest", token.toString());
    }

    @Test
    public void testFormatDurationWithOptionalTokens() {
        String result = DurationFormatUtils.formatDuration(1000L, "H'H'm'M's'S'[optional]", true);
        assertEquals("0H0M1S", result); // Assuming optional part is not included when zero
    }

    @Test
    public void testSuppressLeadingZeroElements() {
        String result = DurationFormatUtils.formatDurationWords(3600000L, true, false);
        assertEquals("1 hour 0 minutes 0 seconds", result);
    }

    @Test
    public void testSuppressTrailingZeroElements() {
        String result = DurationFormatUtils.formatDurationWords(86400000L, false, true);
        assertEquals("1 day", result);
    }

    @Test
    public void testFormatPeriodWithNegativeDifference() {
        long startMillis = System.currentTimeMillis();
        long endMillis = startMillis - 3600000L;
        assertThrows(IllegalArgumentException.class, () -> DurationFormatUtils.formatPeriod(startMillis, endMillis, "H'H'm'M's'S'"));
    }

    @Test
    public void testFormatDurationWithNegativeMilliseconds() {
        assertThrows(IllegalArgumentException.class, () -> DurationFormatUtils.formatDuration(-1000L, "H'H'm'M's'S'"));
    }

    @Test
    public void testAdjustmentsForNegativeValues() {
        Calendar start = Calendar.getInstance();
        Calendar end = Calendar.getInstance();
        start.set(2023, Calendar.OCTOBER, 1, 12, 0, 0);
        end.set(2023, Calendar.OCTOBER, 1, 11, 59, 59);
        String result = DurationFormatUtils.formatPeriod(start.getTimeInMillis(), end.getTimeInMillis(), "H'H'm'M's'S'", true, TimeZone.getDefault());
        assertEquals("23H59M59S", result);
    }

    @Test
    public void testAdjustmentsForNegativeMillisecondsOnly() {
        Calendar start = Calendar.getInstance();
        Calendar end = Calendar.getInstance();
        start.set(2023, Calendar.OCTOBER, 1, 12, 0, 0);
        end.set(2023, Calendar.OCTOBER, 1, 12, 0, 0, -500);
        String result = DurationFormatUtils.formatPeriod(start.getTimeInMillis(), end.getTimeInMillis(), "H'H'm'M's'S'", true, TimeZone.getDefault());
        assertEquals("11H59M59S.500", result);
    }
}
