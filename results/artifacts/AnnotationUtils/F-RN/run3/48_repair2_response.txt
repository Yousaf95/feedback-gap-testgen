package org.apache.commons.lang3;

import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.util.Arrays;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class AnnotationUtils_GenTest {

    @Test
    void testIsValidAnnotationMemberType_withNestedArray() {
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(new Class<?>[] { int[].class }));
    }

    @Test
    void testIsValidAnnotationMemberType_withInvalidArrayComponentType() {
        assertFalse(AnnotationUtils.isValidAnnotationMemberType(new Class<?>[] { Object.class }));
    }

    @Test
    void testEquals_withEmptyAnnotations() {
        TestAnnotation a1 = TestAnnotation.class.getAnnotation(TestAnnotation.class);
        TestAnnotation a2 = TestAnnotation.class.getAnnotation(TestAnnotation.class);
        assertTrue(AnnotationUtils.equals(a1, a2));
    }

    @Test
    void testHashCode_withEmptyAnnotations() {
        TestAnnotation a1 = TestAnnotation.class.getAnnotation(TestAnnotation.class);
        TestAnnotation a2 = TestAnnotation.class.getAnnotation(TestAnnotation.class);
        assertEquals(AnnotationUtils.hashCode(a1), AnnotationUtils.hashCode(a2));
    }

    @Test
    void testToString_withEmptyAnnotations() {
        TestAnnotation a = TestAnnotation.class.getAnnotation(TestAnnotation.class);
        assertEquals("@TestAnnotation(value=\"\")", AnnotationUtils.toString(a));
    }

    @Test
    void testIsValidAnnotationMemberType_withNull() {
        assertFalse(AnnotationUtils.isValidAnnotationMemberType(null));
    }
}

@Retention(RetentionPolicy.RUNTIME)
@interface TestAnnotation {
    String value() default "";
}
