package org.apache.commons.lang3;

import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.util.Arrays;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;
import static org.junit.jupiter.api.Assertions.*;

class AnnotationUtils_GenTest {

    @Retention(RetentionPolicy.RUNTIME)
    public @interface TestAnnotation {
        String value();
        int count() default 0;
    }

    @Retention(RetentionPolicy.RUNTIME)
    public @interface AnotherTestAnnotation {
        String value();
    }

    // ---- added by coverage-guided refinement ----
    @Test
    public void testEquals_withAnnotationsWithDifferentMemberValues() {
        TestAnnotation a1 = create(TestAnnotation.class, "test", 5);
        TestAnnotation a2 = create(TestAnnotation.class, "test", 10);
        assertFalse(AnnotationUtils.equals(a1, a2));
    }

    @Test
    public void testEquals_withAnnotationsWithDifferentMemberTypes() {
        TestAnnotation a1 = create(TestAnnotation.class, "test");
        AnotherTestAnnotation a2 = create(AnotherTestAnnotation.class, "test");
        assertFalse(AnnotationUtils.equals(a1, a2));
    }

    @Test
    public void testHashCode_withAnnotationsWithDifferentMemberValues() {
        TestAnnotation a1 = create(TestAnnotation.class, "test", 5);
        TestAnnotation a2 = create(TestAnnotation.class, "test", 10);
        assertNotEquals(AnnotationUtils.hashCode(a1), AnnotationUtils.hashCode(a2));
    }

    @Test
    public void testHashCode_withAnnotationsWithDifferentMemberTypes() {
        TestAnnotation a1 = create(TestAnnotation.class, "test");
        AnotherTestAnnotation a2 = create(AnotherTestAnnotation.class, "test");
        assertNotEquals(AnnotationUtils.hashCode(a1), AnnotationUtils.hashCode(a2));
    }

    @Test
    public void testIsValidAnnotationMemberType_withNestedArray() {
        Class<?> type = int[][].class;
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(type));
    }

    @Test
    public void testIsValidAnnotationMemberType_withInvalidNestedArray() {
        Class<?> type = int[][][].class;
        assertFalse(AnnotationUtils.isValidAnnotationMemberType(type));
    }

    private static <T extends Annotation> T create(Class<T> annotationClass, Object... values) {
        try {
            T annotation = annotationClass.getConstructor().newInstance();
            for (int i = 0; i < annotation.annotationType().getDeclaredMethods().length; i++) {
                Method method = annotation.annotationType().getDeclaredMethods()[i];
                if (method.getParameterTypes().length == 1) {
                    method.invoke(annotation, values[i]);
                }
            }
            return annotation;
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }
}
