package org.apache.commons.lang3;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;

import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;

import static org.junit.jupiter.api.Assertions.*;

class AnnotationUtils_GenTest {

    @Retention(RetentionPolicy.RUNTIME)
    public @interface TestAnnotation {
        String value() default "testValue";
        int number() default 42;
    }

    @Retention(RetentionPolicy.RUNTIME)
    public @interface AnotherTestAnnotation {
        String name() default "anotherName";
        int count() default 10;
    }

    @Test
    void testToString_nonNullAnnotation() {
        TestAnnotation annotation = new TestAnnotation() {
            @Override
            public String value() {
                return "testValue";
            }

            @Override
            public int number() {
                return 42;
            }
        };
        assertEquals("@TestAnnotation(value=testValue, number=42)", AnnotationUtils.toString(annotation));
    }

    @Test
    void testHashCode_sameInstanceAnnotations() {
        TestAnnotation annotation1 = new TestAnnotation() {
            @Override
            public String value() {
                return "testValue";
            }

            @Override
            public int number() {
                return 42;
            }
        };
        TestAnnotation annotation2 = annotation1;
        assertEquals(AnnotationUtils.hashCode(annotation1), AnnotationUtils.hashCode(annotation2));
    }

    @Test
    void testEquals_identicalAnnotations() {
        TestAnnotation annotation1 = new TestAnnotation() {
            @Override
            public String value() {
                return "testValue";
            }

            @Override
            public int number() {
                return 42;
            }
        };
        TestAnnotation annotation2 = new TestAnnotation() {
            @Override
            public String value() {
                return "testValue";
            }

            @Override
            public int number() {
                return 42;
            }
        };
        assertTrue(AnnotationUtils.equals(annotation1, annotation2));
    }

    @Test
    void testIsValidAnnotationMemberType_arrayOfValidTypes() {
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(String[].class));
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(int[].class));
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(Class[].class));
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(Enum[].class));
        assertTrue(AnnotationUtils.isValidAnnotationMemberType(TestAnnotation[].class));
    }

    @Test
    void testIsValidAnnotationMemberType_arrayOfInvalidTypes() {
        assertFalse(AnnotationUtils.isValidAnnotationMemberType(Object[].class));
        assertFalse(AnnotationUtils.isValidAnnotationMemberType(String[][].class));
    }

    @Test
    void testToString_differentAnnotations() {
        TestAnnotation annotation1 = new TestAnnotation() {
            @Override
            public String value() {
                return "testValue";
            }

            @Override
            public int number() {
                return 42;
            }
        };
        AnotherTestAnnotation annotation2 = new AnotherTestAnnotation() {
            @Override
            public String name() {
                return "anotherName";
            }

            @Override
            public int count() {
                return 10;
            }
        };
        assertNotEquals(AnnotationUtils.toString(annotation1), AnnotationUtils.toString(annotation2));
    }
}
