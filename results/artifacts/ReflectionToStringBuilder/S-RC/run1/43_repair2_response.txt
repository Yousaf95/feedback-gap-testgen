        Object obj = new Object();
        String result = ReflectionToStringBuilder.toString(obj);
        assertNotNull(result);
    }

    @Test
    public void testToStringExcludeFields_2() {
        MyClass obj = new MyClass("value1", null, "value2");
        String result = ReflectionToStringBuilder.toStringExclude(obj, "field2");
        assertTrue(result.contains("field1=value1"));
        assertFalse(result.contains("field2=null"));
        assertTrue(result.contains("field3=value2"));
    }

    @Test
    public void testToStringIncludeFields_2() {
        MyClass obj = new MyClass("value1", null, "value2");
        String result = ReflectionToStringBuilder.toStringInclude(obj, "field2");

        assertTrue(result.contains("field2=null"));
        assertFalse(result.contains("field3=value2"));
    }
