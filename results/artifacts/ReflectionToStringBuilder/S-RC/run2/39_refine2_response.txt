package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Arrays;
import java.util.Collections;

public class ReflectionToStringBuilder_GenTest {

    @Test
    public void testToNoNullStringArrayWithEmptyCollection() {
        String[] result = ReflectionToStringBuilder.toNoNullStringArray(Collections.emptyList());
        assertTrue(Arrays.equals(new String[0], result));
    }

    @Test
    public void testToNoNullStringArrayWithNonEmptyCollection() {
        Collection<String> collection = Arrays.asList("a", "b", null, "c");
        String[] result = ReflectionToStringBuilder.toNoNullStringArray(collection);
        assertTrue(Arrays.equals(new String[]{"a", "b", "c"}, result));
    }

    @Test
    public void testToStringWithEmptyClass() {
        Object object = new Object();
        assertEquals("org.apache.commons.lang3.builder.ReflectionToStringBuilder_GenTest$1@" + Integer.toHexString(object.hashCode()), ReflectionToStringBuilder.toString(object, ToStringStyle.SHORT_PREFIX_STYLE));
    }

    @Test
    public void testToStringWithTransientField_2() throws Exception {
        class TestClass {
            private transient int transientField = 42;
        }
        Object object = new TestClass();
        assertEquals("org.apache.commons.lang3.builder.ReflectionToStringBuilder_GenTest$TestClass@" + Integer.toHexString(object.hashCode()) + "[transientField=42]", ReflectionToStringBuilder.toString(object, ToStringStyle.SHORT_PREFIX_STYLE));
    }

    @Test
    public void testToStringWithStaticField_2() throws Exception {
        class TestClass {
            private static int staticField = 42;
        }
        Object object = new TestClass();
        assertEquals("org.apache.commons.lang3.builder.ReflectionToStringBuilder_GenTest$TestClass@" + Integer.toHexString(object.hashCode()) + "[staticField=42]", ReflectionToStringBuilder.toString(object, ToStringStyle.SHORT_PREFIX_STYLE));
    }

    @Test
    public void testAppendFieldsInWithNonArray() throws Exception {
        class TestClass {
            private int field = 42;
        }
        Object object = new TestClass();
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(object);
        builder.appendFieldsIn(TestClass.class);
        assertEquals("field=42", builder.toString());
    }

    @Test
    public void testGetExcludeFieldNames() {
        String[] excludeFieldNames = {"exclude1", "exclude2"};
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object()).setExcludeFieldNames(excludeFieldNames);
        assertArrayEquals(excludeFieldNames, builder.getExcludeFieldNames());
    }

    @Test
    public void testGetIncludeFieldNames() {
        String[] includeFieldNames = {"include1", "include2"};
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object()).setIncludeFieldNames(includeFieldNames);
        assertArrayEquals(includeFieldNames, builder.getIncludeFieldNames());
    }

    @Test
    public void testIsExcludeNullValues() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object()).setExcludeNullValues(true);
        assertTrue(builder.isExcludeNullValues());
    }

    @Test
    public void testSetUpToClassWithValidClass() throws Exception {
        class TestClass {}
        Object object = new TestClass();
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(object).setUpToClass(TestClass.class);
        assertEquals(TestClass.class, builder.getUpToClass());
    }

    @Test
    public void testValidateWithIntersectingExcludeInclude() throws Exception {
        class TestClass {}
        Object object = new TestClass();
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(object).setExcludeFieldNames("exclude").setIncludeFieldNames("exclude");
        assertThrows(IllegalStateException.class, () -> builder.validate());
    }
}
