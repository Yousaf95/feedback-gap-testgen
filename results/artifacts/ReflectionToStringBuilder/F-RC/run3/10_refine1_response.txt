package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import java.util.Arrays;
import java.util.Collections;

import org.apache.commons.lang3.ArrayUtils;
import org.apache.commons.lang3.builder.ToStringStyle;
import org.junit.jupiter.api.Test;

public class ReflectionToStringBuilder_GenTest {

    @Test
    public void testToNoNullStringArrayWithNullCollection() {
        String[] result = ReflectionToStringBuilder.toNoNullStringArray(null);
        assertArrayEquals(ArrayUtils.EMPTY_STRING_ARRAY, result);
    }

    @Test
    public void testToNoNullStringArrayWithEmptyCollection() {
        Collection<String> emptyCollection = Collections.emptyList();
        String[] result = ReflectionToStringBuilder.toNoNullStringArray(emptyCollection);
        assertArrayEquals(ArrayUtils.EMPTY_STRING_ARRAY, result);
    }

    @Test
    public void testToNoNullStringArrayWithNonEmptyCollection() {
        Collection<String> nonEmptyCollection = Arrays.asList("a", "b", null, "c");
        String[] result = ReflectionToStringBuilder.toNoNullStringArray(nonEmptyCollection);
        assertArrayEquals(new String[]{"a", "b", "c"}, result);
    }

    @Test
    public void testToStringWithDefaultStyle() {
        Object object = new Object();
        String result = ReflectionToStringBuilder.toString(object);
        assertEquals(ToStringStyle.DEFAULT_STYLE.toString(object), result);
    }

    @Test
    public void testToStringWithCustomStyle() {
        Object object = new Object();
        ToStringStyle customStyle = ToStringStyle.SHORT_PREFIX_STYLE;
        String result = ReflectionToStringBuilder.toString(object, customStyle);
        assertEquals(customStyle.toString(object), result);
    }

    @Test
    public void testToStringWithOutputTransients() {
        class TestClass {
            private int transientField = 42;
        }
        TestClass object = new TestClass();
        String result = ReflectionToStringBuilder.toString(object, ToStringStyle.DEFAULT_STYLE, true);
        assertTrue(result.contains("transientField=42"));
    }

    @Test
    public void testToStringWithOutputStatics() {
        class TestClass {
            static int staticField = 42;
        }
        TestClass object = new TestClass();
        String result = ReflectionToStringBuilder.toString(object, ToStringStyle.DEFAULT_STYLE, false, true);
        assertTrue(result.contains("staticField=42"));
    }

    @Test
    public void testToStringWithExcludeNullValues() {
        class TestClass {
            private Integer field = null;
        }
        TestClass object = new TestClass();
        String result = ReflectionToStringBuilder.toString(object, ToStringStyle.DEFAULT_STYLE, false, false, true);
        assertFalse(result.contains("field"));
    }

    @Test
    public void testToStringWithExcludeFieldNames() {
        class TestClass {
            private int field1;
            private int field2;
        }
        TestClass object = new TestClass();
        String result = ReflectionToStringBuilder.toString(object, ToStringStyle.DEFAULT_STYLE, false, false, null, null, "field1");
        assertFalse(result.contains("field1"));
    }

    @Test
    public void testToStringWithIncludeFieldNames() {
        class TestClass {
            private int field1;
            private int field2;
        }
        TestClass object = new TestClass();
        String result = ReflectionToStringBuilder.toString(object, ToStringStyle.DEFAULT_STYLE, false, false, null, null, null, "field1");
        assertTrue(result.contains("field1"));
    }

    @Test
    public void testReflectionAppendArrayWithNull() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.reflectionAppendArray(null);
        assertEquals("", builder.toString());
    }

    @Test
    public void testReflectionAppendArrayWithEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.reflectionAppendArray(ArrayUtils.EMPTY_OBJECT_ARRAY);
        assertEquals("", builder.toString());
    }

    @Test
    public void testReflectionAppendArrayWithNonEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.reflectionAppendArray(new String[]{"a", "b", null, "c"});
        assertTrue(builder.toString().contains("a"));
        assertTrue(builder.toString().contains("b"));
        assertFalse(builder.toString().contains("null"));
        assertTrue(builder.toString().contains("c"));
    }

    @Test
    public void testSetExcludeFieldNamesWithNull() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setExcludeFieldNames(null);
        assertNull(builder.getExcludeFieldNames());
    }

    @Test
    public void testSetExcludeFieldNamesWithEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setExcludeFieldNames(new String[]{});
        assertArrayEquals(ArrayUtils.EMPTY_STRING_ARRAY, builder.getExcludeFieldNames());
    }

    @Test
    public void testSetExcludeFieldNamesWithNonEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setExcludeFieldNames(new String[]{"a", "b", null, "c"});
        assertArrayEquals(new String[]{"a", "b", "c"}, builder.getExcludeFieldNames());
    }

    @Test
    public void testSetIncludeFieldNamesWithNull() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setIncludeFieldNames(null);
        assertNull(builder.getIncludeFieldNames());
    }

    @Test
    public void testSetIncludeFieldNamesWithEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setIncludeFieldNames(new String[]{});
        assertArrayEquals(ArrayUtils.EMPTY_STRING_ARRAY, builder.getIncludeFieldNames());
    }

    @Test
    public void testSetIncludeFieldNamesWithNonEmptyArray() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setIncludeFieldNames(new String[]{"a", "b", null, "c"});
        assertArrayEquals(new String[]{"a", "b", "c"}, builder.getIncludeFieldNames());
    }

    @Test
    public void testSetUpToClassWithNull() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setUpToClass(null);
        assertNull(builder.getUpToClass());
    }

    @Test
    public void testSetUpToClassWithNonSuperclass() {
        class Parent {}
        class Child extends Parent {}
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Child());
        assertThrows(IllegalArgumentException.class, () -> builder.setUpToClass(Parent.class));
    }

    @Test
    public void testIsExcludeNullValues() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        assertFalse(builder.isExcludeNullValues());
        builder.setExcludeNullValues(true);
        assertTrue(builder.isExcludeNullValues());
    }

    @Test
    public void testValidateWithNoConflictingFieldNames() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setExcludeFieldNames(new String[]{"a"});
        builder.setIncludeFieldNames(new String[]{"b"});
        assertDoesNotThrow(() -> builder.validate());
    }

    @Test
    public void testValidateWithConflictingFieldNames() {
        ReflectionToStringBuilder builder = new ReflectionToStringBuilder(new Object());
        builder.setExcludeFieldNames(new String[]{"a"});
        builder.setIncludeFieldNames(new String[]{"a"});
        assertThrows(IllegalStateException.class, () -> builder.validate());
    }
}
