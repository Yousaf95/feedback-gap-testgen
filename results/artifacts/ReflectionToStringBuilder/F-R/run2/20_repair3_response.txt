        assertEquals("null", result);
    }

    @ParameterizedTest
    @ValueSource(strings = {"java.lang.String", "int", "[I"})
    public void testToStringPrimitiveArray(String className) throws Exception {
        Object array;
        if (className.equals("java.lang.String")) {
            array = new String[]{"a", "b", "c"};
        } else if (className.equals("int")) {
            array = new int[]{1, 2, 3};
        } else {
            array = new int[][]{{1, 2}, {3, 4}};
        }
        String result = ReflectionToStringBuilder.toString(array);
        assertEquals(Arrays.deepToString((Object[]) array), result);
    }
