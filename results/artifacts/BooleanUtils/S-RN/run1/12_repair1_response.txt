import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;

class BooleanUtils_GenTest {

    @ParameterizedTest
    @MethodSource("provideBooleans")
    void testToIntegerDefaultIfNull(Boolean bool, int defaultValue) {
        assertEquals(defaultValue, BooleanUtils.toIntegerDefaultIfNull(bool, defaultValue));
    }

    static Stream<Arguments> provideBooleans() {
        return Stream.of(
            Arguments.of(null, 0),
            Arguments.of(Boolean.TRUE, 1),
            Arguments.of(Boolean.FALSE, 0)
        );
    }

    @ParameterizedTest
    @MethodSource("provideBooleans")
    void testToIntegerDefaultIfNullWithNullValue(Boolean bool) {
        assertEquals(0, BooleanUtils.toIntegerDefaultIfNull(bool, 0));
    }

    }

    static Stream<Arguments> provideStringsAndBools() {
        return Stream.of(
            Arguments.of(null, "on", "off", null, null),
            Arguments.of("true", "on", "off", null, "on"),

        );
    }
