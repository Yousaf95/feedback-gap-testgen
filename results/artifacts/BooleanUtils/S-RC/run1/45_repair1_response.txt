package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import static org.junit.jupiter.params.provider.Arguments.of;
import java.util.List;
import java.util.stream.Stream;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;
import java.util.Arrays;

class BooleanUtils_GenTest {

    @ParameterizedTest
    @MethodSource("booleanArrayProvider")
    public void testAnd(boolean[] array, boolean expected) {
        assertEquals(expected, BooleanUtils.and(array));
    }

    @ParameterizedTest
    @MethodSource("booleanArrayProvider")
    public void testOr(boolean[] array, boolean expected) {
        assertEquals(expected, BooleanUtils.or(array));
    }

    @ParameterizedTest
    @MethodSource("booleanArrayProvider")
    public void testOneHot(boolean[] array, boolean expected) {
        assertEquals(expected, BooleanUtils.oneHot(array));
    }

    @Test
    public void testToBooleanObjectNull() {
        assertNull(BooleanUtils.toBooleanObject(null));
    }

    @ParameterizedTest
    @MethodSource("booleanProvider")
    public void testToIntegerObject(boolean value, int expected) {
        assertEquals(expected, BooleanUtils.toIntegerObject(value));
    }

    @ParameterizedTest
    @MethodSource("stringProvider")
    public void testToStringOnOff(String str, String expected) {
        assertEquals(expected, BooleanUtils.toStringOnOff(str));
    }

    @ParameterizedTest
    @MethodSource("booleanProvider")
    public void testToStringTrueFalse(boolean value, String expected) {
        assertEquals(expected, BooleanUtils.toStringTrueFalse(value));
    }

    @ParameterizedTest
    @MethodSource("stringProvider")
    public void testToStringYesNo(String str, String expected) {
        assertEquals(expected, BooleanUtils.toStringYesNo(str));
    }

    private static Stream<Arguments> booleanArrayProvider() {
        return Stream.of(
            of(new boolean[]{}, true),
            of(new boolean[]{true, true, true}, true),
            of(new boolean[]{true, false, true}, false)
        );
    }

    private static Stream<Arguments> booleanProvider() {
        return Stream.of(
            of(true, 1),
            of(false, 0)
        );
    }

    private static Stream<Arguments> stringProvider() {
        return Stream.of(
            of("true", "on"),
            of("false", "off"),
            of(null, null)
        );
    }
}
