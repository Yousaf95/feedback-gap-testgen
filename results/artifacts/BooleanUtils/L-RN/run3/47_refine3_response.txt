package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class BooleanUtils_GenTest {

    @Test
    void testToBooleanObjectWithCustomValuesIncludingNullsAndDefaultTrue() {
        assertEquals(Boolean.TRUE, BooleanUtils.toBooleanObject(1, 1, 0, null));
    }

    @Test
    void testToBooleanObjectWithCustomValuesIncludingNullsAndDefaultFalse() {
        assertEquals(Boolean.FALSE, BooleanUtils.toBooleanObject(0, 1, 0, null));
    }

    @Test
    void testToBooleanObjectWithCustomValuesIncludingNullsAndDefaultValueIsNull() {
        assertNull(BooleanUtils.toBooleanObject(2, 1, 0, 2));
    }

    @Test
    void testToIntegerWithCustomValuesTrueValue() {
        assertEquals(5, BooleanUtils.toInteger(true, 5, 10));
    }

    @Test
    void testToIntegerWithCustomValuesFalseValue() {
        assertEquals(10, BooleanUtils.toInteger(false, 5, 10));
    }

    @Test
    void testToIntegerObjectWithBooleanAndTrueValue() {
        assertEquals(NumberUtils.INTEGER_ONE, BooleanUtils.toIntegerObject(true, NumberUtils.INTEGER_ONE, NumberUtils.INTEGER_ZERO));
    }

    @Test
    void testToIntegerObjectWithBooleanAndFalseValue() {
        assertEquals(NumberUtils.INTEGER_ZERO, BooleanUtils.toIntegerObject(false, NumberUtils.INTEGER_ONE, NumberUtils.INTEGER_ZERO));
    }

    @Test
    void testToStringWithCustomValuesTrueString() {
        assertEquals("Yes", BooleanUtils.toString(true, "Yes", "No"));
    }

    @Test
    void testToStringWithCustomValuesFalseString() {
        assertEquals("No", BooleanUtils.toString(false, "Yes", "No"));
    }

    @Test
    void testForEachExecutesActionOnceForTrue() {
        final boolean[] executed = {false};
        BooleanUtils.forEach(b -> {
            if (b) {
                executed[0] = true;
            }
        });
        assertTrue(executed[0]);
    }

    @Test
    void testForEachExecutesActionOnceForFalse() {
        final boolean[] executed = {false};
        BooleanUtils.forEach(b -> {
            if (!b) {
                executed[0] = true;
            }
        });
        assertTrue(executed[0]);
    }

    @Test
    void testBooleanValuesContainsTrueAndFalse() {
        Boolean[] values = BooleanUtils.booleanValues();
        assertEquals(2, values.length);
        assertTrue(Arrays.asList(values).contains(Boolean.TRUE));
        assertTrue(Arrays.asList(values).contains(Boolean.FALSE));
    }

    @Test
    void testPrimitiveValuesContainsTrueAndFalse() {
        boolean[] values = BooleanUtils.primitiveValues();
        assertEquals(2, values.length);
        assertTrue(Arrays.asList(values).contains(true));
        assertTrue(Arrays.asList(values).contains(false));
    }
}
