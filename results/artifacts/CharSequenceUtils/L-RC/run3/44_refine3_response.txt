package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertThrows;

import org.junit.jupiter.api.Test;

public class CharSequenceUtils_GenTest {

    @Test
    void testSubSequence_withStartIndexLessThanZero() {
        assertEquals("abc", CharSequenceUtils.subSequence("abc", -1));
    }

    @Test
    void testToCharArray_withNonEmptyStringBuilder() {
        char[] expected = {'a', 'b', 'c'};
        assertEquals(java.util.Arrays.toString(expected), java.util.Arrays.toString(CharSequenceUtils.toCharArray(new StringBuilder("abc"))));
    }

    @Test
    void testToCharArray_withNonEmptyStringBuffer() {
        char[] expected = {'a', 'b', 'c'};
        assertEquals(java.util.Arrays.toString(expected), java.util.Arrays.toString(CharSequenceUtils.toCharArray(new StringBuffer("abc"))));
    }

    @Test
    void testIndexOf_withNonSpecificInstance() {
        assertEquals(1, CharSequenceUtils.indexOf(new StringBuilder("abc"), "b"));
    }

    @Test
    void testIndexOf_withStartLessThanZeroAndMatch() {
        assertEquals(0, CharSequenceUtils.indexOf(new StringBuilder("abc"), 'a', -1));
    }

    @Test
    void testIndexOf_withSupplementaryCharacterMatch() {
        int supplementaryChar = 0x1F600; // Grinning Face Emoji
        assertEquals(0, CharSequenceUtils.indexOf(Character.toChars(supplementaryChar), supplementaryChar));
    }

    @Test
    void testLastIndexOf_withNonSpecificInstance_2() {
        assertEquals(1, CharSequenceUtils.lastIndexOf(new StringBuilder("abc"), "b"));
    }

    @Test
    void testLastIndexOf_withStartGreaterThanLength_2() {
        assertEquals(2, CharSequenceUtils.lastIndexOf(new StringBuilder("abc"), 'c', 5));
    }

    @Test
    void testLastIndexOf_withEmptySearchChar() {
        assertThrows(StringIndexOutOfBoundsException.class, () -> CharSequenceUtils.lastIndexOf("abc", "", 1));
    }

    @Test
    void testLastIndexOf_withStartLessThanZero() {
        assertEquals(-1, CharSequenceUtils.lastIndexOf(new StringBuilder("abc"), 'a', -1));
    }

    @Test
    void testLastIndexOf_withSupplementaryCharacterMatch() {
        int supplementaryChar = 0x1F600; // Grinning Face Emoji
        assertEquals(0, CharSequenceUtils.lastIndexOf(Character.toChars(supplementaryChar), supplementaryChar));
    }
}
