package org.apache.commons.lang3;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class CharSequenceUtils_GenTest {

    @Test
    void testIndexOf_StartLessThanZero() {
        assertEquals(0, CharSequenceUtils.indexOf("abc", "a", -1));
    }

    @Test
    void testIndexOf_SupplementaryCharacterNotFound() {
        assertEquals(-1, CharSequenceUtils.indexOf("\uD835\uDC9C", "\uD835\uDC9D"));
    }

    @Test
    void testIndexOf_StringAndSearchSequence_NotFound() {
        assertEquals(-1, CharSequenceUtils.indexOf("abc", "d"));
    }

    @Test
    void testLastIndexOf_StartLessThanZero() {
        assertEquals(-1, CharSequenceUtils.lastIndexOf("abc", "a", -1));
    }

    @Test
    void testLastIndexOf_SupplementaryCharacterNotFound() {
        assertEquals(-1, CharSequenceUtils.lastIndexOf("\uD835\uDC9C", "\uD835\uDC9D"));
    }

    @Test
    void testLastIndexOf_StringAndSearchSequence_NotFound() {
        assertEquals(-1, CharSequenceUtils.lastIndexOf("abc", "d"));
    }
}
