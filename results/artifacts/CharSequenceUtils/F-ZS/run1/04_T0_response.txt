package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import static org.apache.commons.lang3.StringUtils.*;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;

public class CharSequenceUtils_GenTest {

    @ParameterizedTest
    @ValueSource(ints = {0, 1, 5})
    public void testSubSequence_withValidStartIndex(CharSequence cs, int start) {
        CharSequence result = CharSequenceUtils.subSequence(cs, start);
        assertEquals(cs.subSequence(start, cs.length()), result);
    }

    @ParameterizedTest
    @ValueSource(ints = {-1, -2, Integer.MAX_VALUE})
    public void testSubSequence_withInvalidStartIndex(CharSequence cs, int start) {
        assertThrows(IndexOutOfBoundsException.class, () -> CharSequenceUtils.subSequence(cs, start));
    }

    @Test
    public void testSubSequence_withNull() {
        assertNull(CharSequenceUtils.subSequence(null, 0));
    }

    @ParameterizedTest
    @ValueSource(strings = {"", "a", "abc"})
    public void testToCharArray_withValidString(CharSequence cs) {
        char[] result = CharSequenceUtils.toCharArray(cs);
        assertArrayEquals(toCharArray(cs), result);
    }

    @Test
    public void testToCharArray_withNull() {
        char[] result = CharSequenceUtils.toCharArray(null);
        assertArrayEquals(ArrayUtils.EMPTY_CHAR_ARRAY, result);
    }
}
