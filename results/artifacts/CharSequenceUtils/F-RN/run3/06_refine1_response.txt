package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharSequenceUtils_GenTest {

    @Test
    public void testSubSequence_NullInput() {
        assertThrows(NullPointerException.class, () -> CharSequenceUtils.subSequence(null, 0));
    }

    @Test
    public void testSubSequence_EmptyString() {
        assertEquals("", CharSequenceUtils.subSequence("", 0));
    }

    @Test
    public void testSubSequence_StartIndexTooLarge() {
        assertThrows(IndexOutOfBoundsException.class, () -> CharSequenceUtils.subSequence("abc", 3));
    }

    @Test
    public void testSubSequence_StartIndexNegative() {
        assertThrows(IndexOutOfBoundsException.class, () -> CharSequenceUtils.subSequence("abc", -1));
    }

    @Test
    public void testToCharArray_NullInput() {
        assertThrows(NullPointerException.class, () -> CharSequenceUtils.toCharArray(null));
    }

    @Test
    public void testToCharArray_EmptyString() {
        char[] result = CharSequenceUtils.toCharArray("");
        assertNotNull(result);
        assertEquals(0, result.length);
    }

    @Test
    public void testToCharArray_NormalString() {
        char[] result = CharSequenceUtils.toCharArray("abc");
        assertNotNull(result);
        assertArrayEquals(new char[]{'a', 'b', 'c'}, result);
    }
}
