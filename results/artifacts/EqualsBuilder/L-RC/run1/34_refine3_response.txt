import java.util.Set;

    void testRegistryWithNull() {
        Set<Pair<IDKey, IDKey>> registry = null;
        REGISTRY.set(registry);
        assertEquals(0, getRegistry().size());
    }

    @Test
    void testRegisterWithNullRegistry() {
        REGISTRY.remove();
        register(new Object(), new Object());
        assertNotNull(getRegistry());
    }

    @Test
    void testUnregisterWithEmptyRegistry() {
        REGISTRY.set(new HashSet<>());
        unregister(new Object(), new Object());
        assertTrue(getRegistry().isEmpty());
    }
