package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Arrays;
import java.util.Collections;

public class EqualsBuilder_GenTest {

    @Test
    void testReflectionEqualsWithSameObject() {
        Object obj = new Object();
        assertTrue(EqualsBuilder.reflectionEquals(obj, obj));
    }

    @Test
    void testReflectionEqualsWithNullObjects() {
        assertFalse(EqualsBuilder.reflectionEquals(null, null, false, null));
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), null, false, null));
        assertFalse(EqualsBuilder.reflectionEquals(null, new Object(), false, null));
    }

    @Test
    void testReflectionEqualsWithDifferentObjects() {
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), new Object()));
    }

    @Test
    void testReflectionEqualsWithBooleanParameter() {
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), new Object(), true));
    }

    @Test
    void testReflectionEqualsWithClassParameter() {
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), new Object(), false, String.class));
    }

    @Test
    void testReflectionEqualsWithExcludeFields() {
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), new Object(), "field1"));
    }

    @Test
    void testReflectionEqualsWithCollectionExcludeFields() {
        assertFalse(EqualsBuilder.reflectionEquals(new Object(), new Object(), Collections.singletonList("field1")));
    }

    @Test
    void testAppendBooleanWhenNotEqual() {
        EqualsBuilder builder = new EqualsBuilder().append(true, false);
        assertFalse(builder.isEquals());
    }

    @Test
    void testAppendBooleanArrayWithNulls() {
