        EqualsBuilder.register(new Object(), new Object());
        assertNotNull(REGISTRY.get());
    }

    void testUnregisterWithNonNullRegistry() {
        REGISTRY.set(new HashSet<>()); // Set a non-null registry
        Object obj1 = new Object();
        Object obj2 = new Object();
        EqualsBuilder.register(obj1, obj2);
        assertEquals(1, REGISTRY.get().size());
        EqualsBuilder.unregister(obj1, obj2);
        assertTrue(REGISTRY.get().isEmpty());
    }
