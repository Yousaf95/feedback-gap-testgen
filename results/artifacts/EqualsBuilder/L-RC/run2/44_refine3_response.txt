public class EqualsBuilder_GenTest {

    @Test
    void testReflectionEqualsWithNullObjects() {
        assertFalse(EqualsBuilder.reflectionEquals(null, null, true));
    }

    @Test
    void testReflectionEqualsWithSameObjectReference() {
        Object obj = new Object();
        assertTrue(EqualsBuilder.reflectionEquals(obj, obj, false));
    }

    @Test
    void testRegisterWithNullRegistry() {
        REGISTRY.set(null);
        EqualsBuilder.register(new Object(), new Object());
        assertNotNull(REGISTRY.get());
    }

    @Test

        Set<Pair<IDKey, IDKey>> registry = new HashSet<>();
        REGISTRY.set(registry);
        EqualsBuilder.unregister(new Object(), new Object());
        assertNull(REGISTRY.get());
    }

    @Test
