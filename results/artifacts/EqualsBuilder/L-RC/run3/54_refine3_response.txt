package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.HashSet;
import java.util.Set;

public class EqualsBuilder_GenTest {

    @Test
    void testRegistryContainsPair() {
        Set<Pair<IDKey, IDKey>> registry = new HashSet<>();
        Pair<IDKey, IDKey> pair = Pair.of(new IDKey("test1"), new IDKey("test2"));
        registry.add(pair);
        assertEquals(true, EqualsBuilder.containsInRegistry(pair, registry));
    }

    @Test
    void testRegistryContainsSwappedPair() {
        Set<Pair<IDKey, IDKey>> registry = new HashSet<>();
        Pair<IDKey, IDKey> pair = Pair.of(new IDKey("test1"), new IDKey("test2"));
        Pair<IDKey, IDKey> swappedPair = Pair.of(pair.getRight(), pair.getLeft());
        registry.add(swappedPair);
        assertEquals(true, EqualsBuilder.containsInRegistry(pair, registry));
    }

    @Test
    void testRegistryNull() {
        Pair<IDKey, IDKey> pair = Pair.of(new IDKey("test1"), new IDKey("test2"));
        assertEquals(false, EqualsBuilder.containsInRegistry(pair, null));
    }

    @Test
    void testRegisterWithNullRegistry() {
        EqualsBuilder.register(null, null);
        assertNotNull(EqualsBuilder.getRegistry());
    }

    @Test
    void testUnregisterWithEmptyRegistry() {
        Set<Pair<IDKey, IDKey>> registry = new HashSet<>();
        REGISTRY.set(registry);
        EqualsBuilder.unregister(new Object(), new Object());
        assertNull(REGISTRY.get());
    }

    @Test
    void testAppendBooleanWhenNotEquals() {
        EqualsBuilder builder = new EqualsBuilder();
        builder.isEquals = false;
        assertEquals(builder, builder.append(true, false));
    }

    @Test
    void testAppendBooleanArrayWithDifferentLengths() {
        boolean[] lhs = {true};
        boolean[] rhs = {};
        EqualsBuilder builder = new EqualsBuilder().append(lhs, rhs);
        assertFalse(builder.isEquals());
    }

    @Test
    void testAppendByteArrayWithDifferentLengths() {
        byte[] lhs = {1};

        assertFalse(builder.isEquals());
    }

    }
}
