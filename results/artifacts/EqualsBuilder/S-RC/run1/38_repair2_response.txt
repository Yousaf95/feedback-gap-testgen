package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import static org.junit.jupiter.params.provider.Arguments.of;
import java.util.Arrays;
import java.util.List;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;

class EqualsBuilder_GenTest {

    private EqualsBuilder equalsBuilder;

    @BeforeEach
    void setUp() {
        equalsBuilder = new EqualsBuilder();
    }

    @ParameterizedTest
    @MethodSource("provideObjects")
    void testEquals(Object lhs, Object rhs) {
        boolean result = EqualsBuilder.reflectionEquals(lhs, rhs);
        assertEquals(result, lhs.equals(rhs));
    }

    private static Stream<Arguments> provideObjects() {
        return Arrays.stream(new Object[][]{
            {new Object(), new Object()},
            {"test", "test"},
            {123, 123},
            {true, true}
        }).map(Arguments::of);
    }
}
