package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import java.util.Comparator;
import java.util.function.Function;
import org.junit.jupiter.api.Test;

public class Range_GenTest {

    @Test
    public void testFitWithElementOutsideRange() {
        Range<Integer> range = Range.between(1, 5);
        assertEquals(1, range.fit(0));
        assertEquals(5, range.fit(6));
    }

    @Test
    public void testIsOverlappedByWithNonOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 3);
        Range<Integer> range2 = Range.between(4, 6);
        assertFalse(range1.isOverlappedBy(range2));
        assertFalse(range2.isOverlappedBy(range1));
    }

    @Test
    public void testIsOverlappedByWithPartiallyOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 5);
        Range<Integer> range2 = Range.between(4, 8);
        assertTrue(range1.isOverlappedBy(range2));
        assertTrue(range2.isOverlappedBy(range1));
    }

    @Test
    public void testIsOverlappedByWithFullyOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 5);
        Range<Integer> range2 = Range.between(3, 4);
        assertTrue(range1.isOverlappedBy(range2));
        assertTrue(range2.isOverlappedBy(range1));
    }

    @Test
    public void testIntersectionWithNonOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 3);
        Range<Integer> range2 = Range.between(4, 6);
        assertThrows(IllegalArgumentException.class, () -> range1.intersectionWith(range2));
    }

    @Test
    public void testIntersectionWithPartiallyOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 5);
        Range<Integer> range2 = Range.between(4, 8);
        assertEquals(Range.between(4, 5), range1.intersectionWith(range2));
    }

    @Test
    public void testIntersectionWithFullyOverlappingRanges() {
        Range<Integer> range1 = Range.between(1, 5);
        Range<Integer> range2 = Range.between(3, 4);
        assertEquals(Range.between(3, 4), range1.intersectionWith(range2));
    }

    @Test
    public void testIsNaturalOrderingWithDefaultComparator() {
        Range<Integer> range = Range.between(1, 5);
        assertTrue(range.isNaturalOrdering());
    }

    @Test
    public void testIsNaturalOrderingWithCustomComparator() {
        Comparator<String> customComparator = Comparator.reverseOrder();
        Range<String> range = Range.between("a", "c", customComparator);
        assertFalse(range.isNaturalOrdering());
    }
}
