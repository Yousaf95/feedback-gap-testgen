package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import java.util.Comparator;
import java.util.function.Function;
import org.junit.jupiter.api.Test;

public class Range_GenTest {

    @Test
    public void testOfWithComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range = Range.of(1, 5, comparator);
        assertEquals(Integer.valueOf(1), range.getMinimum());
        assertEquals(Integer.valueOf(5), range.getMaximum());
        assertEquals(comparator, range.getComparator());
    }

    @Test
    public void testContainsRangeNonOverlapping() {
        Range<Integer> range1 = Range.of(1, 3);
        Range<Integer> range2 = Range.of(4, 6);
        assertFalse(range1.containsRange(range2));
    }

    @Test
    public void testElementCompareToNull() {
        Range<Integer> range = Range.of(1, 5);
        assertThrows(NullPointerException.class, () -> range.elementCompareTo(null));
    }

    @Test
    public void testFitBelow() {
        Range<Integer> range = Range.of(3, 7);
        assertEquals(Integer.valueOf(3), range.fit(2));
    }

    @Test
    public void testFitAbove() {
        Range<Integer> range = Range.of(3, 7);
        assertEquals(Integer.valueOf(7), range.fit(8));
    }

    @Test
    public void testHashCodeSameComparators() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.of(1, 5, comparator);
        Range<Integer> range2 = Range.of(1, 5, comparator);
        assertEquals(range1.hashCode(), range2.hashCode());
    }

    @Test
    public void testIntersectionWithNonOverlapping() {
        Range<Integer> range1 = Range.of(1, 3);
        Range<Integer> range2 = Range.of(4, 6);
        assertThrows(IllegalArgumentException.class, () -> range1.intersectionWith(range2));
    }

    @Test
    public void testIsAfterNull() {
        Range<Integer> range = Range.of(1, 5);
        assertFalse(range.isAfter(null));
    }

    @Test
    public void testIsBeforeNull() {
        Range<Integer> range = Range.of(1, 5);
        assertFalse(range.isBefore(null));
    }

    @Test
    public void testIsEndedByNull() {
        Range<Integer> range = Range.of(1, 5);
        assertFalse(range.isEndedBy(null));
    }

    @Test
    public void testIsStartedByNull() {
        Range<Integer> range = Range.of(1, 5);
        assertFalse(range.isStartedBy(null));
    }

    @Test
    public void testToStringFormat() {
        Range<Integer> range = Range.of(1, 5);
        assertEquals("[1..5]", range.toString("%s"));
    }
}
