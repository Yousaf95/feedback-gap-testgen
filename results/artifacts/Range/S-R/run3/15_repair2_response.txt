package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import static org.junit.jupiter.params.provider.Arguments.arguments;
import java.util.Comparator;
import java.util.stream.Stream;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;

public class Range_GenTest {

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testRangeCreation(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals(from, range.getMinimum());
        assertEquals(to, range.getMaximum());
        if (comparator == null) {
            assertTrue(range.isNaturalOrdering());
        } else {
            assertNotSame(Comparator.naturalOrder(), range.getComparator());
        }
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testContains(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.contains(from));
        assertTrue(range.contains(to));
        assertFalse(range.contains(Comparator.naturalOrder().compare(from, to) > 0 ? to : from));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testContainsRange(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.containsRange(Range.of(from, to, comparator)));
        assertFalse(range.containsRange(Range.of(Comparator.naturalOrder().compare(from, to) > 0 ? to : from, from, comparator)));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testElementCompareTo(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals(0, range.elementCompareTo(from));
        assertEquals(0, range.elementCompareTo(to));
        assertTrue(range.elementCompareTo(Comparator.naturalOrder().compare(from, to) > 0 ? to : from) < 0);
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testFit(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals(from, range.fit(Comparator.naturalOrder().compare(from, to) > 0 ? to : from));
        assertEquals(to, range.fit(Comparator.naturalOrder().compare(from, to) < 0 ? from : to));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testGetComparator(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        if (comparator == null) {
            assertTrue(range.isNaturalOrdering());
        } else {
            assertNotSame(Comparator.naturalOrder(), range.getComparator());
        }
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testGetMaximum(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals(to, range.getMaximum());
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testGetMinimum(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals(from, range.getMinimum());
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testHashCode(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        int hashCode = 17;
        hashCode = 37 * hashCode + getClass().hashCode();
        hashCode = 37 * hashCode + range.getMinimum().hashCode();
        hashCode = 37 * hashCode + range.getMaximum().hashCode();
        assertEquals(hashCode, range.hashCode());
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testIsAfter(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.isAfter(Comparator.naturalOrder().compare(from, to) > 0 ? to : from));
        assertFalse(range.isAfter(to));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testIsBefore(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertFalse(range.isBefore(from));
        assertTrue(range.isBefore(Comparator.naturalOrder().compare(from, to) < 0 ? to : from));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testIsEndedBy(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.isEndedBy(to));
        assertFalse(range.isEndedBy(from));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testIsOverlappedBy(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.isOverlappedBy(Range.of(from, to, comparator)));
        assertFalse(range.isOverlappedBy(Range.of(Comparator.naturalOrder().compare(from, to) > 0 ? to : from, from, comparator)));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testIsStartedBy(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertTrue(range.isStartedBy(from));
        assertFalse(range.isStartedBy(to));
    }

    @ParameterizedTest
    @MethodSource("provideRangeCreationData")
    public <T extends Comparable<? super T>> void testToString(T from, T to, Comparator<T> comparator) {
        Range<T> range = Range.of(from, to, comparator);
        assertEquals("[" + range.getMinimum() + ".." + range.getMaximum() + "]", range.toString());
    }

    private static Stream<Arguments> provideRangeCreationData() {
        return Stream.of(
            arguments(1, 5, null),
            arguments(10.5, 20.5, Comparator.naturalOrder()),
            arguments("a", "z", Comparator.reverseOrder())
        );
    }
}
