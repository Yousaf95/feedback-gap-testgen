package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class Range_GenTest {

    @Test
    void testIntersectionWithIdenticalRangesUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(1, 5, comparator);
        Range<Integer> range2 = Range.between(1, 5, comparator);
        Range<Integer> intersection = range1.intersectionWith(range2);
        assertEquals(Range.between(1, 5), intersection);
    }

    @Test
    void testIntersectionWithOverlappingStartUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(1, 5, comparator);
        Range<Integer> range2 = Range.between(3, 7, comparator);
        Range<Integer> intersection = range1.intersectionWith(range2);
        assertEquals(Range.between(3, 5), intersection);
    }

    @Test
    void testIntersectionWithOverlappingEndUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(3, 7, comparator);
        Range<Integer> range2 = Range.between(1, 5, comparator);
        Range<Integer> intersection = range1.intersectionWith(range2);
        assertEquals(Range.between(3, 5), intersection);
    }

    @Test
    void testIsNaturalOrderingWithNullComparator() {
        Range<Integer> range = Range.of(1, 5, null);
        assertTrue(range.isNaturalOrdering());
    }

    @Test
    void testFitElementWithinRangeUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range = Range.between(3, 7, comparator);
        assertEquals(Integer.valueOf(5), range.fit(5));
    }

    @Test
    void testFitElementBelowRangeUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range = Range.between(3, 7, comparator);
        assertEquals(Integer.valueOf(3), range.fit(1));
    }

    @Test
    void testFitElementAboveRangeUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range = Range.between(3, 7, comparator);
        assertEquals(Integer.valueOf(7), range.fit(9));
    }

    @Test
    void testIsStartedByWithNullComparator() {
        Range<Integer> range = Range.of(1, 5, null);
        assertTrue(range.isStartedBy(1));
    }

    @Test
    void testIsEndedByWithNullComparator() {
        Range<Integer> range = Range.of(1, 5, null);
        assertTrue(range.isEndedBy(5));
    }

    @Test
    void testIsAfterRangeNonOverlappingUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(1, 3, comparator);
        Range<Integer> range2 = Range.between(4, 6, comparator);
        assertTrue(range2.isAfterRange(range1));
    }

    @Test
    void testIsBeforeRangeNonOverlappingUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(4, 6, comparator);
        Range<Integer> range2 = Range.between(1, 3, comparator);
        assertTrue(range2.isBeforeRange(range1));
    }

    @Test
    void testIntersectionWithNonOverlappingThrowsExceptionUsingComparator() {
        Comparator<Integer> comparator = Integer::compareTo;
        Range<Integer> range1 = Range.between(1, 3, comparator);
        Range<Integer> range2 = Range.between(4, 6, comparator);
        assertThrows(IllegalArgumentException.class, () -> range1.intersectionWith(range2));
    }
}
