package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import java.util.Comparator;
import java.util.function.Function;

public class Range_GenTest {

    @Test
    public void testIsEmptyRange() {
        Range<Integer> range = Range.between(5, 4);
        assertTrue(range.isEmpty());
    }

    @Test
    public void testIsEmptyRangeWithComparator() {
        Comparator<Integer> reverseOrder = Comparator.reverseOrder();
        Range<Integer> range = Range.between(5, 4, reverseOrder);
        assertTrue(range.isEmpty());
    }

    @Test
    public void testIsNotEmptyRange() {
        Range<Integer> range = Range.between(1, 3);
        assertFalse(range.isEmpty());
    }

    @Test
    public void testIsNotEmptyRangeWithComparator() {
        Comparator<Integer> reverseOrder = Comparator.reverseOrder();
        Range<Integer> range = Range.between(3, 1, reverseOrder);
        assertFalse(range.isEmpty());
    }

    @Test
    public void testIntersectionWithEmptyRangeThrowsException() {
        Range<Integer> range = Range.between(1, 5);
        assertThrows(IllegalArgumentException.class, () -> range.intersectionWith(Range.is(6)));
    }

    @Test
    public void testIsOverlappedByNonOverlappingRange() {
        Range<Integer> range = Range.between(1, 5);
        assertFalse(range.isOverlappedBy(Range.between(6, 10)));
    }

    @Test
    public void testIsNotOverlappedByNonOverlappingRangeWithComparator() {
        Comparator<Integer> reverseOrder = Comparator.reverseOrder();
        Range<Integer> range = Range.between(1, 5, reverseOrder);
        assertFalse(range.isOverlappedBy(Range.between(6, 10, reverseOrder)));
    }

    @Test
    public void testIsOverlappedByPartiallyOverlappingRange() {
        Range<Integer> range = Range.between(1, 5);
        assertTrue(range.isOverlappedBy(Range.between(3, 7)));
    }

    @Test
    public void testIsNotOverlappedByPartiallyOverlappingRangeWithComparator() {
        Comparator<Integer> reverseOrder = Comparator.reverseOrder();
        Range<Integer> range = Range.between(1, 5, reverseOrder);
        assertTrue(range.isOverlappedBy(Range.between(7, 3, reverseOrder)));
    }

    @Test
    public void testIsOverlappedByFullyOverlappingRange() {
        Range<Integer> range = Range.between(1, 5);
        assertTrue(range.isOverlappedBy(Range.between(2, 4)));
    }

    @Test
    public void testIsNotOverlappedByFullyOverlappingRangeWithComparator() {
        Comparator<Integer> reverseOrder = Comparator.reverseOrder();
        Range<Integer> range = Range.between(1, 5, reverseOrder);
        assertTrue(range.isOverlappedBy(Range.between(4, 2, reverseOrder)));
    }

    @Test
    public void testIsNotOverlappedByNonOverlappingRangeWithNullComparator() {
        Range<Integer> range = Range.between(1, 5, null);
        assertFalse(range.isOverlappedBy(Range.between(6, 10, null)));
    }
}
