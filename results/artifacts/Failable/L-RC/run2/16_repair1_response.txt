        biConsumer.accept("test", "value");

    void testAsBiFunctionNormalCase() {
        BiFunction<String, String, String> biFunction = Failable.asBiFunction((input1, input2) -> input1 + input2);
        assertTrue(biFunction.apply("Hello", "World").equals("HelloWorld"));
    }

    void testAsBiPredicateNormalCase() {
        BiPredicate<String, String> biPredicate = Failable.asBiPredicate((input1, input2) -> input1.equals(input2));
