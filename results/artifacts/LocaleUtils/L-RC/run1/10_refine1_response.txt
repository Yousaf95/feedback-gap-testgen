package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.List;
import java.util.Locale;

public class LocaleUtils_GenTest {

    @Test
    void testSyncAvoidInitialization() {
        assertNotNull(LocaleUtils.availableLocaleList());
        assertNotNull(LocaleUtils.availableLocaleSet());
    }

    @Test
    void testIsISO3166CountryCode_withValidCode() {
        assertTrue(LocaleUtils.isISO3166CountryCode("US"));
        assertFalse(LocaleUtils.isISO3166CountryCode("USA"));
    }

    @Test
    void testIsISO3166CountryCode_withInvalidCode() {
        assertFalse(LocaleUtils.isISO3166CountryCode("USa"));
        assertFalse(LocaleUtils.isISO3166CountryCode("U"));
        assertFalse(LocaleUtils.isISO3166CountryCode(""));
    }

    @Test
    void testIsISO639LanguageCode_withValidCodes() {
        assertTrue(LocaleUtils.isISO639LanguageCode("en"));
        assertTrue(LocaleUtils.isISO639LanguageCode("eng"));
    }

    @Test
    void testIsISO639LanguageCode_withInvalidCodes() {
        assertFalse(LocaleUtils.isISO639LanguageCode("engg"));
        assertFalse(LocaleUtils.isISO639LanguageCode("e"));
        assertFalse(LocaleUtils.isISO639LanguageCode(""));
        assertFalse(LocaleUtils.isISO639LanguageCode("123"));
    }

    @Test
    void testIsNumericAreaCode_withValidCode() {
        assertTrue(LocaleUtils.isNumericAreaCode("123"));
    }

    @Test
    void testIsNumericAreaCode_withInvalidCodes() {
        assertFalse(LocaleUtils.isNumericAreaCode("12a"));
        assertFalse(LocaleUtils.isNumericAreaCode("1234"));
        assertFalse(LocaleUtils.isNumericAreaCode(""));
    }

    @Test
    void testLocaleLookupList_withNullLocale() {
        List<Locale> result = LocaleUtils.localeLookupList(null);
        assertEquals(1, result.size());
        assertEquals(Locale.getDefault(), result.get(0));
    }

    @Test
    void testParseLocale_withLanguageOnly() {
        Locale locale = LocaleUtils.parseLocale("en");
        assertEquals(new Locale("en"), locale);
    }

    @Test
    void testParseLocale_withLanguageAndCountry() {
        Locale locale = LocaleUtils.parseLocale("en_US");
        assertEquals(new Locale("en", "US"), locale);
    }

    @Test
    void testParseLocale_withLanguageCountryAndVariant() {
        Locale locale = LocaleUtils.parseLocale("en_US_WIN");
        assertEquals(new Locale("en", "US", "WIN"), locale);
    }

    @Test
    void testParseLocale_withInvalidFormat() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("en_US_"));
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("en_US_WIN32"));
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("_US"));
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("-US"));
    }

    @Test
    void testParseLocale_withEmptyString() {
        assertEquals(new Locale("", ""), LocaleUtils.parseLocale(""));
    }

    @Test
    void testParseLocale_withHashCharacter() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("en_US#WIN"));
    }

    @Test
    void testParseLocale_withShortCountryCode() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("en_U"));
    }

    @Test
    void testParseLocale_withInvalidSeparator() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.parseLocale("en-US-WIN"));
    }

    @Test
    void testToString_withHashCharacter() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.toLocale("en_US#WIN"));
    }

    @Test
    void testToString_withShortCountryCode() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.toLocale("en_U"));
    }

    @Test
    void testToString_withInvalidSeparator() {
        assertThrows(IllegalArgumentException.class, () -> LocaleUtils.toLocale("en-US-WIN"));
    }

    @Test
    void testLocaleConstructor() {
        new LocaleUtils();
    }
}
