package org.apache.commons.lang3;

import java.util.Locale;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class LocaleUtils_GenTest {

    @Test
    public void testCountriesByLanguageWithValidInput() {
        List<Locale> countries = LocaleUtils.countriesByLanguage("en");
        assertTrue(countries.stream().anyMatch(locale -> "US".equals(locale.getCountry())));
        assertTrue(countries.stream().anyMatch(locale -> "GB".equals(locale.getCountry())));
    }

    @Test
    public void testLanguagesByCountryWithValidInput() {
        List<Locale> languages = LocaleUtils.languagesByCountry("US");
        assertTrue(languages.stream().anyMatch(locale -> "en".equals(locale.getLanguage())));
    }

    @Test
    public void testToLocaleWithScriptAndExtensions_1() {
        Locale locale = LocaleUtils.toLocale("zh-Hans-CN");
        assertEquals("zh", locale.getLanguage());
        assertEquals("CN", locale.getCountry());
        assertNull(locale.getVariant());
    }

    @Test
    public void testToLocaleWithScriptAndExtensions_2() {
        Locale locale = LocaleUtils.toLocale("fr-FR-x-private");
        assertEquals("fr", locale.getLanguage());
        assertEquals("FR", locale.getCountry());
        assertNull(locale.getVariant());
    }

    @Test
    public void testIsLanguageUndeterminedWithValidLocale() {
        Locale locale = new Locale("und", "US");
        assertTrue(LocaleUtils.isLanguageUndetermined(locale));
    }

    @Test
    public void testIsLanguageUndeterminedWithUndeterminedLocale() {
        Locale locale = new Locale("und", "");
        assertTrue(LocaleUtils.isLanguageUndetermined(locale));
    }

    @Test
    public void testToLocaleWithValidFormat_1() {
        Locale locale = LocaleUtils.toLocale("de_DE");
        assertEquals("de", locale.getLanguage());
        assertEquals("DE", locale.getCountry());
        assertNull(locale.getVariant());
    }

    @Test
    public void testToLocaleWithValidFormat_2() {
        Locale locale = LocaleUtils.toLocale("it_IT");
        assertEquals("it", locale.getLanguage());
        assertEquals("IT", locale.getCountry());
        assertNull(locale.getVariant());
    }
}
