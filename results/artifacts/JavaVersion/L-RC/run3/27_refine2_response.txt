package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNull;

import org.junit.jupiter.api.Test;

public class JavaVersion_GenTest {

    @Test
    public void testGetWithVersion0_9Point5() {
        assertEquals(JavaVersion.JAVA_RECENT, JavaVersion.get("0.9.5"));
    }

    @Test
    public void testGetWithVersion1_8Point9() {
        assertEquals(JavaVersion.JAVA_RECENT, JavaVersion.get("1.8.9"));
    }

    @Test
    public void testGetWithVersion10Point5() {
        assertEquals(JavaVersion.JAVA_RECENT, JavaVersion.get("10.5"));
    }

    @Test
    public void testGetWithVersion22() {
        assertEquals(JavaVersion.JAVA_RECENT, JavaVersion.get("22"));
    }

    @Test
    public void testGetWithVersion9Point0() {
        assertEquals(JavaVersion.JAVA_9, JavaVersion.get("9.0"));
    }

    @Test
    public void testGetWithVersion1_0() {
        assertNull(JavaVersion.get("1.0"));
    }

    @Test
    public void testMaxVersionWithDefaultProperty() {
        System.setProperty("java.specification.version", "99.0");
        assertEquals(99.0f, JavaVersion.maxVersion());
    }

    @Test
    public void testMaxVersionWithNegativeProperty() {
        System.setProperty("java.specification.version", "-1.0");
        assertEquals(99.0f, JavaVersion.maxVersion());
    }

    @Test
    public void testToFloatVersionWithSingleDigit() {
        assertEquals(21.0f, JavaVersion.toFloatVersion("21"));
    }

    @Test
    public void testToFloatVersionWithNoDot() {
        assertEquals(99.0f, JavaVersion.toFloatVersion("99"));
    }
}
