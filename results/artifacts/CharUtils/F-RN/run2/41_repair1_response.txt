import static org.junit.jupiter.api.Assertions.*;
import static org.apache.commons.lang3.CharUtils.*;

import java.util.List;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;

class CharUtils_GenTest {

    @Test
    public void testIsAsciiControl_ExtendedAscii() {
        assertFalse(isAsciiControl('\u007F')); // DEL character
    }

    @Test
    public void testIsAsciiNonPrintable_ControlCharacters() {
        assertTrue(isAsciiNonPrintable('\t')); // HT tab
        assertTrue(isAsciiNonPrintable('\n')); // LF linefeed
        assertTrue(isAsciiNonPrintable('\r')); // CR carriage return
        assertTrue(isAsciiNonPrintable('\b')); // BS backspace
    }

    @Test
    public void testIsAsciiNonPrintable_ExtendedAscii() {
        assertTrue(isAsciiNonPrintable('\u0080')); // Extended ASCII character
    }

    @Test
    public void testToChar_CharacterNull() {
        assertThrows(NullPointerException.class, () -> toChar(null));
    }

    @Test
    public void testToChar_StringEmpty() {
        assertThrows(IllegalArgumentException.class, () -> toChar(""));
    }

    @Test
    public void testToIntValue_InvalidCharacter_2() {
        assertThrows(IllegalArgumentException.class, () -> toIntValue('a'));
    }

    @Test
    public void testUnicodeEscaped_CharacterExtendedAscii() {
        assertEquals("\\u00A1", unicodeEscaped('\u00A1')); // Inverted exclamation mark
    }

    @Test
    public void testToString_CharacterNull() {
        assertNull(toString(null));
    }
}
