package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharUtils_GenTest {

    @Test
    public void testIsAsciiControl_ExtendedAscii() {
        assertFalse(CharUtils.isAsciiControl('\u007F')); // DEL character
    }

    @Test
    public void testIsAsciiNonPrintable_ControlCharacters() {
        assertTrue(CharUtils.isAsciiNonPrintable('\t')); // HT tab
        assertTrue(CharUtils.isAsciiNonPrintable('\n')); // LF linefeed
        assertTrue(CharUtils.isAsciiNonPrintable('\r')); // CR carriage return
        assertTrue(CharUtils.isAsciiNonPrintable('\b')); // BS backspace
    }

    @Test
    public void testIsAsciiNonPrintable_ExtendedAscii() {
        assertTrue(CharUtils.isAsciiNonPrintable('\u0080')); // Extended ASCII character
    }

    @Test
    public void testToChar_CharacterNull() {
        assertThrows(NullPointerException.class, () -> CharUtils.toChar(null));
    }

    @Test
    public void testToChar_StringEmpty() {
        assertThrows(IllegalArgumentException.class, () -> CharUtils.toChar(""));
    }

    @Test
    public void testToIntValue_InvalidCharacter() {
        assertThrows(IllegalArgumentException.class, () -> CharUtils.toIntValue('a'));
    }

    @Test
    public void testUnicodeEscaped_CharacterExtendedAscii() {
        assertEquals("\\u00A1", CharUtils.unicodeEscaped('\u00A1')); // Inverted exclamation mark
    }

    @Test
    public void testToString_CharacterNull() {
        assertNull(CharUtils.toString(null));
    }
}
