package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharUtils_GenTest {

    @Test
    public void testIsAsciiControl() {
        assertTrue(CharUtils.isAsciiControl('\0'));
        assertTrue(CharUtils.isAsciiControl('\t'));
        assertTrue(CharUtils.isAsciiControl('\n'));
        assertTrue(CharUtils.isAsciiControl('\r'));
        assertTrue(CharUtils.isAsciiControl('\f'));
        assertTrue(CharUtils.isAsciiControl('\b'));
        assertFalse(CharUtils.isAsciiControl('a'));
    }

    @Test
    public void testIsAsciiNonPrintable() {
        assertTrue(CharUtils.isAsciiNonPrintable('\0'));
        assertTrue(CharUtils.isAsciiNonPrintable('\t'));
        assertTrue(CharUtils.isAsciiNonPrintable('\n'));
        assertTrue(CharUtils.isAsciiNonPrintable('\r'));
        assertTrue(CharUtils.isAsciiNonPrintable('\f'));
        assertTrue(CharUtils.isAsciiNonPrintable('\b'));
        assertFalse(CharUtils.isAsciiNonPrintable('a'));
    }

    @Test
    public void testIsAsciiAlpha() {
        assertTrue(CharUtils.isAsciiAlpha('a'));
        assertTrue(CharUtils.isAsciiAlpha('z'));
        assertTrue(CharUtils.isAsciiAlpha('A'));
        assertTrue(CharUtils.isAsciiAlpha('Z'));
        assertFalse(CharUtils.isAsciiAlpha('0'));
        assertFalse(CharUtils.isAsciiAlpha('9'));
    }

    @Test
    public void testIsAsciiNumeric() {
        assertTrue(CharUtils.isAsciiNumeric('0'));
        assertTrue(CharUtils.isAsciiNumeric('9'));
        assertFalse(CharUtils.isAsciiNumeric('a'));
        assertFalse(CharUtils.isAsciiNumeric('z'));
        assertFalse(CharUtils.isAsciiNumeric('A'));
        assertFalse(CharUtils.isAsciiNumeric('Z'));
    }

    @Test
    public void testIsAsciiAlphanumeric() {
        assertTrue(CharUtils.isAsciiAlphanumeric('0'));
        assertTrue(CharUtils.isAsciiAlphanumeric('9'));
        assertTrue(CharUtils.isAsciiAlphanumeric('a'));
        assertTrue(CharUtils.isAsciiAlphanumeric('z'));
        assertTrue(CharUtils.isAsciiAlphanumeric('A'));
        assertTrue(CharUtils.isAsciiAlphanumeric('Z'));
        assertFalse(CharUtils.isAsciiAlphanumeric(' '));
    }

    @Test
    public void testToIntValue_InvalidCharacter() {
        assertThrows(IllegalArgumentException.class, () -> CharUtils.toIntValue('a'));
    }

    @Test
    public void testToString_Character() {
        assertEquals("a", CharUtils.toString('a'));
        assertEquals("\u0020", CharUtils.toString(' '));
    }

    @Test
    public void testToString_CharacterObject() {
        assertEquals("a", CharUtils.toString(Character.valueOf('a')));
        assertEquals("\u0020", CharUtils.toString(Character.valueOf(' ')));
    }
}
