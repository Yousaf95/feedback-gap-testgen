import static org.junit.jupiter.api.Assertions.*;
import static org.junit.jupiter.params.provider.Arguments.arguments;
import java.util.stream.Stream;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.Arguments;
import org.junit.jupiter.params.provider.MethodSource;

class CharUtils_GenTest {

    @ParameterizedTest
    @MethodSource("asciiExtendedProvider")
    public void testIsAsciiExtended_2(char ch, boolean expected) {
        assertEquals(expected, CharUtils.isAscii(ch));
    }

    private static Stream<Arguments> asciiExtendedProvider() {
        return Stream.of(
            arguments('\u007F', true),
            arguments('\u0080', false)
        );
    }

    }
