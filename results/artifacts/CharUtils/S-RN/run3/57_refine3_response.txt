package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharUtils_GenTest {

    @Test
    public void testIsAsciiExtended() {
        assertTrue(CharUtils.isAsciiExtended((char) 128));
        assertFalse(CharUtils.isAsciiExtended((char) 255));
    }

    @Test
    public void testIsAsciiAlphaExtended() {
        assertTrue(CharUtils.isAsciiAlphaExtended('€'));
        assertFalse(CharUtils.isAsciiAlphaExtended('a'));
    }

    @Test
    public void testIsAsciiAlphanumericExtended() {
        assertTrue(CharUtils.isAsciiAlphanumericExtended('$'));
        assertFalse(CharUtils.isAsciiAlphanumericExtended('a'));
    }

    @Test
    public void testIsAsciiNumericExtended() {
        assertTrue(CharUtils.isAsciiNumericExtended('*'));
        assertFalse(CharUtils.isAsciiNumericExtended('0'));
    }

    @Test
    public void testIsAsciiPrintableExtended() {
        assertTrue(CharUtils.isAsciiPrintableExtended((char) 127));
        assertFalse(CharUtils.isAsciiPrintableExtended((char) 256));
    }

    @Test
    public void testToString_CharacterExtended() {
        assertEquals("\\u0080", CharUtils.toString((char) 128));
        assertEquals("€", CharUtils.toString('€'));
    }
}
