package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharUtils_GenTest {

    @Test
    public void testIsAsciiExtended() {
        assertTrue(CharUtils.isAsciiExtended('\u007F'));
        assertFalse(CharUtils.isAsciiExtended('\u0080'));
    }

    @Test
    public void testIsAsciiAlphaExtended() {
        assertTrue(CharUtils.isAsciiAlphaExtended('A'));
        assertTrue(CharUtils.isAsciiAlphaExtended('z'));
        assertFalse(CharUtils.isAsciiAlphaExtended('\u007F'));
        assertFalse(CharUtils.isAsciiAlphaExtended('\u0100')));
    }

    @Test
    public void testIsAsciiAlphanumericExtended() {
        assertTrue(CharUtils.isAsciiAlphanumericExtended('A'));
        assertTrue(CharUtils.isAsciiAlphanumericExtended('z'));
        assertTrue(CharUtils.isAsciiAlphanumericExtended('0'));
        assertFalse(CharUtils.isAsciiAlphanumericExtended('\u007F'));
        assertFalse(CharUtils.isAsciiAlphanumericExtended('\u0100')));
    }

    @Test
    public void testIsAsciiNumericExtended() {
        assertTrue(CharUtils.isAsciiNumericExtended('0'));
        assertTrue(CharUtils.isAsciiNumericExtended('9'));
        assertFalse(CharUtils.isAsciiNumericExtended('\u007F'));
        assertFalse(CharUtils.isAsciiNumericExtended('\u0100')));
    }

    @Test
    public void testIsAsciiPrintableExtended() {
        assertTrue(CharUtils.isAsciiPrintableExtended('A'));
        assertTrue(CharUtils.isAsciiPrintableExtended('z'));
        assertTrue(CharUtils.isAsciiPrintableExtended('0'));
        assertTrue(CharUtils.isAsciiPrintableExtended('\u007F'));
        assertFalse(CharUtils.isAsciiPrintableExtended('\u0100')));
    }

    @Test
    public void testToChar_CharacterNull() {
        assertThrows(NullPointerException.class, () -> CharUtils.toChar(null));
    }

    @Test
    public void testToCharacterObject_StringEmpty() {
        assertNull(CharUtils.toCharacterObject(""));
    }

    @Test
    public void testToIntValue_CharacterNonNumeric() {
        assertThrows(IllegalArgumentException.class, () -> CharUtils.toIntValue('a'));
    }

    @Test
    public void testToIntValue_CharacterNonNumericDefaultValue() {
        assertEquals(42, CharUtils.toIntValue('a', 42));
    }

    @Test
    public void testUnicodeEscaped_CharacterExtended() {
        assertEquals("\\u0100", CharUtils.unicodeEscaped('\u0100'));
    }
}
