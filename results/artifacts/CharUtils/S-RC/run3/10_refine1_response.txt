package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class CharUtils_GenTest {

    @Test
    public void testIsAsciiAlphaLower_UncoveredBranches() {
        assertFalse(CharUtils.isAsciiAlphaLower('A')); // ASCII upper case letter
        assertTrue(CharUtils.isAsciiAlphaLower('z'));  // ASCII lower case letter
        assertFalse(CharUtils.isAsciiAlphaLower('0')); // Non-alphabetic character
    }

    @Test
    public void testIsAsciiControl_UncoveredBranches() {
        assertTrue(CharUtils.isAsciiControl('\t')); // Horizontal tab control character
        assertFalse(CharUtils.isAsciiControl(' '));  // Space character
        assertTrue(CharUtils.isAsciiControl(127));  // Delete control character
    }

    @Test
    public void testIsAsciiPrintable_UncoveredBranches() {
        assertTrue(CharUtils.isAsciiPrintable('A')); // ASCII upper case letter
        assertFalse(CharUtils.isAsciiPrintable('\t')); // Horizontal tab control character
        assertTrue(CharUtils.isAsciiPrintable(127));  // Delete control character
    }

    @Test
    public void testToChar_CharacterNonNull() {
        assertEquals('A', CharUtils.toChar(Character.valueOf('A')));
    }

    @Test
    public void testToCharacterObject_Characters() {
        assertEquals(Character.valueOf('A'), CharUtils.toCharacterObject('A'));
    }

    @Test
    public void testToIntValue_CharDefault() {
        assertEquals(5, CharUtils.toIntValue('5', 0));
    }

    @Test
    public void testToString_Character() {
        assertNull(CharUtils.toString(null));
    }

    @Test
    public void testUnicodeEscaped_Character() {
        assertEquals("\\u0041", CharUtils.unicodeEscaped(Character.valueOf('A')));
    }

    @Test
    public void testUnicodeEscaped_Characters() {
        assertEquals("\\u0041", CharUtils.unicodeEscaped('A'));
    }
}
