public class CharUtils_GenTest {

        assertTrue(CharUtils.isAsciiAlphaLower('a'));

    public void testIsAsciiAlphaLower_UpperCase() {

    }

    }

    }

    public void testIsAsciiPrintable_PrintableCharacter() {

    }
