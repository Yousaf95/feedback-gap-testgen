    @Test
    void testReflectionCompareExcludeFieldsCollection() {
        class TestClass {
            public int field1;
            public transient int field2;

            public TestClass(int field1, int field2) {
                this.field1 = field1;
                this.field2 = field2;
            }
        }

        TestClass obj1 = new TestClass(1, 2);
        TestClass obj2 = new TestClass(1, 3);

        assertEquals(0, CompareToBuilder.reflectionCompare(obj1, obj2, Collections.singleton("field2")));
    }

    @Test
    void testReflectionCompareWithTransients() {

        }

        assertEquals(0, CompareToBuilder.reflectionCompare(obj1, obj2, true));
    }

    @Test
    void testReflectionCompareWithReflectUpToClass() {
        class ParentClass {
            public int parentField;
        }

        class ChildClass extends ParentClass {
            public int childField;

            public ChildClass(int parentField, int childField) {
                this.parentField = parentField;
                this.childField = childField;
            }
        }

        ChildClass obj1 = new ChildClass(1, 2);
        ChildClass obj2 = new ChildClass(1, 3);

        assertEquals(0, CompareToBuilder.reflectionCompare(obj1, obj2, false, ParentClass.class));
    }
