import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Arrays;

public class CompareToBuilder_GenTest {

    @Test
    public void testAppendNull() {
        CompareToBuilder builder = new CompareToBuilder();
        builder.append((Object) null, (Object) null);
        assertEquals(0, builder.toComparison());
    }

    @Test
    public void testAppendNullWithComparator() {
        CompareToBuilder builder = new CompareToBuilder();
        assertThrows(ClassCastException.class, () -> builder.append((Object) null, "string", Comparator.naturalOrder()));
    }

    @Test
    public void testAppendArrayWithNullElements() {
        Integer[] lhs = {1, 2, null};

        builder.append(lhs, rhs);
        assertEquals(0, builder.toComparison());
    }

        assertEquals(-1, builder.toComparison());
    }
