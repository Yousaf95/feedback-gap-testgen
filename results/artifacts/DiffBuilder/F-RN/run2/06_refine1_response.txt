    @Test
    public void testAppendNull() {
        DiffResult<String> result = new DiffBuilder<>(null, "value", ToStringStyle.SHORT_PREFIX_STYLE).append("field", null, "value").build();
        assertEquals(1, result.getDiffs().size());
        assertEquals("field", result.getDiffs().get(0).getFieldName());
        assertNull(result.getDiffs().get(0).getLeft());
        assertEquals("value", result.getDiffs().get(0).getRight());
    }
