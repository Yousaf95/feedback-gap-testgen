package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import java.util.Arrays;
import java.util.List;
import org.junit.jupiter.api.Test;

public class DiffBuilder_GenTest {

    @Test
    public void testAppendShort() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, 2, ToStringStyle.DEFAULT_STYLE);
        builder.append("value", (short) 1, (short) 3);
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertEquals(1, diffs.size());
        assertEquals((short) 1, diffs.get(0).getLeft());
        assertEquals((short) 3, diffs.get(0).getRight());
    }

    @Test
    public void testAppendShortArray() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, 2, ToStringStyle.DEFAULT_STYLE);
        builder.append("value", new short[]{1}, new short[]{3});
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertEquals(1, diffs.size());
        assertArrayEquals(new Short[]{1}, (Short[]) diffs.get(0).getLeft());
        assertArrayEquals(new Short[]{3}, (Short[]) diffs.get(0).getRight());
    }

    @Test
    public void testAppendNull() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, null, ToStringStyle.DEFAULT_STYLE);
        builder.append("value", 1, null);
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertEquals(1, diffs.size());
        assertNull(diffs.get(0).getLeft());
        assertEquals(1, diffs.get(0).getRight());
    }

    @Test
    public void testAppendTriviallyEqualObjects() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, 1, ToStringStyle.DEFAULT_STYLE);
        builder.append("value", 1, 1);
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertTrue(diffs.isEmpty());
    }

    @Test
    public void testAppendObjectArray() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, 2, ToStringStyle.DEFAULT_STYLE);
        builder.append("value", new Integer[]{1}, new Integer[]{3});
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertEquals(1, diffs.size());
        assertArrayEquals(new Integer[]{1}, (Integer[]) diffs.get(0).getLeft());
        assertArrayEquals(new Integer[]{3}, (Integer[]) diffs.get(0).getRight());
    }

    @Test
    public void testAppendDiffResult() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, 2, ToStringStyle.DEFAULT_STYLE);
        DiffBuilder<Integer> nestedBuilder = new DiffBuilder<>(1, 3, ToStringStyle.DEFAULT_STYLE);
        nestedBuilder.append("nested", 1, 4);
        builder.append("value", nestedBuilder.build());
        List<Diff<?>> diffs = builder.build().getDiffs();
        assertEquals(1, diffs.size());
        assertEquals(1, ((Diff<DiffResult<Integer>>) diffs.get(0)).getLeft().getDiffs().size());
    }

    @Test
    public void testAppendBooleanWithNull() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, null, ToStringStyle.DEFAULT_STYLE);
        assertThrows(NullPointerException.class, () -> builder.append("value", true, null));
    }

    @Test

        assertThrows(NullPointerException.class, () -> builder.append("value", 'a', null));

    @Test
    public void testAppendDoubleWithNull() {
        DiffBuilder<Integer> builder = new DiffBuilder<>(1, null, ToStringStyle.DEFAULT_STYLE);
