import static org.junit.jupiter.api.Assertions.*;
import static org.apache.commons.lang3.builder.DiffBuilder.*;

import java.util.Arrays;
import java.util.List;

public class DiffBuilder_GenTest {

    @Test
    public void testAppendShortArray_3() {
        DiffResult<Integer> result = diff(1, 2)
                .append("array", new short[]{1, 2}, new short[]{1, 3})
                .build();
        
        assertEquals(1, result.getDiffs().size());
        assertEquals("array[1]", result.getDiffs().get(0).getFieldName());
        assertEquals((short) 2, result.getDiffs().get(0).getLeft());
        assertEquals((short) 3, result.getDiffs().get(0).getRight());
    }

    @Test
    public void testAppendObjectArray_3() {
        DiffResult<String> result = diff("a", "b")
                .append("array", new String[]{"a", "b"}, new String[]{"a", "c"})

        assertEquals(1, result.getDiffs().size());
        assertEquals("array[1]", result.getDiffs().get(0).getFieldName());
        assertEquals("b", result.getDiffs().get(0).getLeft());
        assertEquals("c", result.getDiffs().get(0).getRight());
