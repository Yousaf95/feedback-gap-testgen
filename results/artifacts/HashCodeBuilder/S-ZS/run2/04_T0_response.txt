package org.apache.commons.lang3.builder;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class HashCodeBuilder_GenTest {

    @Test
    public void testHashCodeBuilderDefaultConstructor() {
        HashCodeBuilder builder = new HashCodeBuilder();
        assertEquals(17, builder.iTotal);
        assertEquals(37, builder.iConstant);
    }

    @Test
    public void testHashCodeBuilderCustomConstructor() {
        HashCodeBuilder builder = new HashCodeBuilder(23, 47);
        assertEquals(23, builder.iTotal);
        assertEquals(47, builder.iConstant);
    }

    @Test
    public void testAppendBoolean() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(true);
        assertEquals(17 * 37 + 0, builder.toHashCode());
        builder.append(false);
        assertEquals(17 * 37 * 37 + 1, builder.toHashCode());
    }

    @Test
    public void testAppendBooleanArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new boolean[]{true, false, true});
        assertEquals(17 * 37 * 37 * 37 * 37 + (0 * 37 * 37 * 37) + (1 * 37 * 37) + (0 * 37) + (1 * 1), builder.toHashCode());
    }

    @Test
    public void testAppendByte() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((byte) 5);
        assertEquals(17 * 37 + 5, builder.toHashCode());
    }

    @Test
    public void testAppendByteArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new byte[]{10, 20, 30});
        assertEquals(17 * 37 * 37 * 37 * 37 + (10 * 37 * 37 * 37) + (20 * 37 * 37) + (30 * 37), builder.toHashCode());
    }

    @Test
    public void testAppendChar() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append('A');
        assertEquals(17 * 37 + 'A', builder.toHashCode());
    }

    @Test
    public void testAppendCharArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new char[]{'a', 'b', 'c'});
        assertEquals(17 * 37 * 37 * 37 * 37 + ('a' * 37 * 37 * 37) + ('b' * 37 * 37) + ('c' * 37), builder.toHashCode());
    }

    @Test
    public void testAppendDouble() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(1.23);
        assertEquals(17 * 37 + Double.doubleToLongBits(1.23), builder.toHashCode());
    }

    @Test
    public void testAppendDoubleArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new double[]{0.5, 1.0, 1.5});
        assertEquals(17 * 37 * 37 * 37 * 37 + Double.doubleToLongBits(0.5) * 37 * 37 * 37 + Double.doubleToLongBits(1.0) * 37 * 37 + Double.doubleToLongBits(1.5), builder.toHashCode());
    }

    @Test
    public void testAppendFloat() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(2.718f);
        assertEquals(17 * 37 + Float.floatToIntBits(2.718f), builder.toHashCode());
    }

    @Test
    public void testAppendFloatArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new float[]{0.1f, 0.2f, 0.3f});
        assertEquals(17 * 37 * 37 * 37 * 37 + Float.floatToIntBits(0.1f) * 37 * 37 * 37 + Float.floatToIntBits(0.2f) * 37 * 37 + Float.floatToIntBits(0.3f), builder.toHashCode());
    }

    @Test
    public void testAppendInt() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(42);
        assertEquals(17 * 37 + 42, builder.toHashCode());
    }

    @Test
    public void testAppendIntArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new int[]{100, 200, 300});
        assertEquals(17 * 37 * 37 * 37 * 37 + (100 * 37 * 37 * 37) + (200 * 37 * 37) + (300 * 37), builder.toHashCode());
    }

    @Test
    public void testAppendLong() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(123456789L);
        assertEquals(17 * 37 + (int) (123456789L ^ 123456789L >> 32), builder.toHashCode());
    }

    @Test
    public void testAppendLongArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new long[]{1000L, 2000L, 3000L});
        assertEquals(17 * 37 * 37 * 37 * 37 + (int) (1000L ^ 1000L >> 32) * 37 * 37 * 37 + (int) (2000L ^ 2000L >> 32) * 37 * 37 + (int) (3000L ^ 3000L >> 32), builder.toHashCode());
    }

    @Test
    public void testAppendObject() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new Object());
        assertEquals(17 * 37 + System.identityHashCode(new Object()), builder.toHashCode());
    }

    @Test
    public void testAppendObjectArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new Object[]{new Object(), new Object()});
        assertEquals(17 * 37 * 37 * 37 * 37 + System.identityHashCode(new Object()) * 37 * 37 * 37 + System.identityHashCode(new Object()), builder.toHashCode());
    }

    @Test
    public void testAppendShort() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((short) 10);
        assertEquals(17 * 37 + 10, builder.toHashCode());
    }

    @Test
    public void testAppendShortArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new short[]{5, 10, 15});
        assertEquals(17 * 37 * 37 * 37 * 37 + (5 * 37 * 37 * 37) + (10 * 37 * 37) + (15 * 37), builder.toHashCode());
    }

    @Test
    public void testAppendSuper() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.appendSuper(42);
        assertEquals(17 * 37 + 42, builder.toHashCode());
    }

    @Test
    public void testBuild() {
        HashCodeBuilder builder = new HashCodeBuilder();
        Integer hashCode = builder.build();
        assertNotNull(hashCode);
        assertEquals(builder.toHashCode(), hashCode.intValue());
    }

    @Test
    public void testEquals() {
        HashCodeBuilder builder1 = new HashCodeBuilder(23, 47).append(1).append(2);
        HashCodeBuilder builder2 = new HashCodeBuilder(23, 47).append(1).append(2);
        assertTrue(builder1.equals(builder2));
    }

    @Test
    public void testHashCode() {
        HashCodeBuilder builder = new HashCodeBuilder();
        assertEquals(builder.toHashCode(), builder.hashCode());
    }
}
