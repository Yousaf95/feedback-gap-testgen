package org.apache.commons.lang3.builder;

import java.util.Arrays;
import java.util.Collection;
import java.util.Comparator;
import java.util.List;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class HashCodeBuilder_GenTest {

    @Test
    public void testAppendEmptyCollection() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((Collection<?>) null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyCollection() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(Arrays.asList("a", "b", "c"));
        assertEquals(208364595, builder.toHashCode());
    }

    @Test
    public void testAppendNullObjectArray() {
