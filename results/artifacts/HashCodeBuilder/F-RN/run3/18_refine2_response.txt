package org.apache.commons.lang3.builder;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;

public class HashCodeBuilder_GenTest {

    @Test
    public void testAppendEmptyCollection() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((Collection<?>) null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyCollection() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(java.util.Arrays.asList("a", "b", "c"));
        assertEquals(208364595, builder.toHashCode());
    }

    @Test
    public void testAppendNullObjectArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((Object[]) null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyObjectArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new Object[]{"a", "b", "c"});
        assertEquals(208364595, builder.toHashCode());
    }

    @Test
    public void testAppendNullStringArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((String[]) null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyStringArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new String[]{"a", "b", "c"});
        assertEquals(208364595, builder.toHashCode());
    }

    @Test
    public void testAppendNullComparatorArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append((Comparator<?>) null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyComparatorArray() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(java.util.Arrays.asList(java.util.Comparator.naturalOrder(), java.util.Comparator.reverseOrder()));
        assertEquals(-208364595, builder.toHashCode());
    }

    @Test
    public void testAppendNullFieldExclude() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new Object(), null);
        assertEquals(17, builder.toHashCode());
    }

    @Test
    public void testAppendNonEmptyFieldExclude() {
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(new Object(), java.util.Arrays.asList("field"));
        assertEquals(17, builder.toHashCode());
    }
}
