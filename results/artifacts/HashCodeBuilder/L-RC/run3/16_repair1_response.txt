package org.apache.commons.lang3.builder;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;
import java.util.Arrays;
import java.util.Collections;

class HashCodeBuilder_GenTest {

    @Test
    void testGetRegistry() {
        // This will indirectly cover getRegistry()
        HashCodeBuilder.reflectionHashCode(new Object());
    }

    @Test
    void testIsRegistered() {
        // Removed register and unregister calls as they are private
        Object obj = new Object();
        assertFalse(HashCodeBuilder.isRegistered(obj));
    }

    @Test
    void testReflectionAppend() {
        class TestClass {
            private int value = 42;
        }
        TestClass obj = new TestClass();
        HashCodeBuilder builder = new HashCodeBuilder();
        // Removed reflectionAppend call as it is private
        assertEquals(17 * 37 + 42, HashCodeBuilder.reflectionHashCode(obj));
    }

    @Test
    void testReflectionHashCodeWithInitialAndMultiplier() {
        class TestClass {
            private int value = 42;
        }
        TestClass obj = new TestClass();
        int hashCode = HashCodeBuilder.reflectionHashCode(13, 29, obj);
        assertEquals(13 * 29 + 42, hashCode);
    }

    @Test
    void testReflectionHashCodeWithTransients() {
        class TestClass {
            private transient int value = 42;
        }
        TestClass obj = new TestClass();
        int hashCode = HashCodeBuilder.reflectionHashCode(obj, true);
        assertEquals(17 * 37, hashCode); // Transient field is included
    }

    @Test
    void testReflectionHashCodeWithExcludeFields() {
        class TestClass {
            private int value = 42;
            private int excludedValue = 84;
        }
        TestClass obj = new TestClass();
        int hashCode = HashCodeBuilder.reflectionHashCode(obj, "excludedValue");
        assertEquals(17 * 37 + 42, hashCode); // Excluded field is not included
    }

    @Test
    void testReflectionHashCodeWithExcludeFieldsCollection() {
        class TestClass {
            private int value = 42;
            private int excludedValue = 84;
        }
        TestClass obj = new TestClass();
        int hashCode = HashCodeBuilder.reflectionHashCode(obj, Collections.singleton("excludedValue"));
        assertEquals(17 * 37 + 42, hashCode); // Excluded field is not included
    }

    @Test
    void testRegisterAndUnregister() {
        // Removed register and unregister calls as they are private
        Object obj = new Object();
        assertFalse(HashCodeBuilder.isRegistered(obj));
    }

    @Test
    void testAppendArrayWithBooleanArray() {
        boolean[] array = {true, false};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 0 * 37 + 1, builder.toHashCode());
    }

    @Test
    void testAppendArrayWithByteArray() {
        byte[] array = {1, 2};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 1 * 37 + 2, builder.toHashCode());
    }

    @Test
    void testAppendArrayWithCharArray() {
        char[] array = {'a', 'b'};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 'a' * 37 + 'b', builder.toHashCode());
    }

    @Test
    void testAppendArrayWithDoubleArray() {
        double[] array = {1.0, 2.0};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + Long.hashCode(Double.doubleToLongBits(1.0)) * 37 + Long.hashCode(Double.doubleToLongBits(2.0)), builder.toHashCode());
    }

    @Test
    void testAppendArrayWithFloatArray() {
        float[] array = {1.0f, 2.0f};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + Float.floatToIntBits(1.0f) * 37 + Float.floatToIntBits(2.0f), builder.toHashCode());
    }

    @Test
    void testAppendArrayWithIntArray() {
        int[] array = {1, 2};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 1 * 37 + 2, builder.toHashCode());
    }

    @Test
    void testAppendArrayWithLongArray() {
        long[] array = {1L, 2L};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + (int) (1L ^ 1L >> 32) * 37 + (int) (2L ^ 2L >> 32), builder.toHashCode());
    }

    @Test
    void testAppendArrayWithObjectArray() {
        Object[] array = {new Integer(1), new Integer(2)};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 1 * 37 + 2, builder.toHashCode());
    }

    @Test
    void testAppendArrayWithShortArray() {
        short[] array = {1, 2};
        HashCodeBuilder builder = new HashCodeBuilder();
        builder.append(array);
        assertEquals(17 * 37 + 1 * 37 + 2, builder.toHashCode());
    }
}
