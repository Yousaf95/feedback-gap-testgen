    @Test
    void testIsRegistered_ObjectNotRegistered_ReturnsFalse() {
        assertFalse(HashCodeBuilder.isRegistered(new Object()));
    }

    @Test
    void testReflectionAppend_ObjectRegistered_ReturnsEarly() {
        Set<IDKey> registry = new HashSet<>();
        registry.add(new IDKey(new Object()));
        HashCodeBuilder.REGISTRY.set(registry);
        HashCodeBuilder reflectionAppendSpy = new HashCodeBuilder();
        HashCodeBuilder.reflectionAppend(new Object(), Object.class, reflectionAppendSpy, false, null);
        assertEquals(HashCodeBuilder.DEFAULT_INITIAL_VALUE, reflectionAppendSpy.toHashCode());
    }

    @Test
    void testReflectionHashCode_DefaultValues() {

        assertEquals(2905984, result); // Expected hash code based on default values and fields of TestObject
    }
