import static org.junit.jupiter.api.Assertions.*;

class ObjectUtils_GenTest {

    @Test
        void testHashCodeHex() {
            assertEquals("0", ObjectUtils.hashCodeHex(null));
            assertEquals(Integer.toHexString(Objects.hashCode("test")), ObjectUtils.hashCodeHex("test"));
        }

    @ParameterizedTest
    @MethodSource("provideIsEmptyData")
    void testIsEmpty_2(String input, boolean expected) {
        assertEquals(expected, ObjectUtils.isEmpty(input));
        assertEquals(expected, ObjectUtils.isEmpty(new ArrayList<>())); // empty list
        assertEquals(expected, ObjectUtils.isEmpty((Object) null));
    }

    private static Stream<Arguments> provideIsEmptyData() {
        return Stream.of(
            of("", true),
            of("value", false)
        );
    }

    @ParameterizedTest
    @MethodSource("provideIsNotEmptyData")
    void testIsNotEmpty(String input, boolean expected) {
        assertEquals(expected, ObjectUtils.isNotEmpty(input));
        assertEquals(expected, ObjectUtils.isNotEmpty(new ArrayList<>(Collections.singletonList("value")))); // non-empty list
        assertEquals(expected, ObjectUtils.isNotEmpty((Object) null));
    }

    private static Stream<Arguments> provideIsNotEmptyData() {
        return Stream.of(
            of("", false),
            of("value", true)
        );
    }

    @Test
        void testMedian() {
            assertEquals(2.0, ObjectUtils.median(1.0, 2.0, 3.0));
            assertEquals(2.0, ObjectUtils.median((Comparator<Double>) Double::compareTo, 1.0, 2.0, 3.0));
        }

    @Test
        void testWait() throws InterruptedException {
        // This method is not easily testable without a real object and thread.
        // We can only ensure that it does not throw an exception.

        ObjectUtils.wait(obj, duration);
    }
