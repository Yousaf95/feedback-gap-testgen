    @ParameterizedTest
    @MethodSource("provideFirstNonNullCases")
    public <T> void testFirstNonNull(T... values) {
        assertEquals(ObjectUtils.firstNonNull(values), Arrays.stream(values).filter(Objects::nonNull).findFirst().orElse(null));
    }

    private static Stream<Arguments> provideFirstNonNullCases() {
        return Stream.of(
            of(1, null, 3),
            of("a", "b", null),
            of(true, false, true)
        );
    }

    @ParameterizedTest
    @MethodSource("provideGetFirstNonNullCases")
    public <T> void testGetFirstNonNull(Supplier<T>... suppliers) {
        assertEquals(ObjectUtils.getFirstNonNull(suppliers), Arrays.stream(suppliers).map(Supplier::get).filter(Objects::nonNull).findFirst().orElse(null));
    }

    private static Stream<Arguments> provideGetFirstNonNullCases() {
        return Stream.of(
            of(() -> 1, () -> null, () -> 3),
            of(() -> "a", () -> "b", () -> null),
            of(() -> true, () -> false, () -> true)
        );
    }

    @ParameterizedTest
    @MethodSource("provideGetIfNullCases")
    public <T> void testGetIfNull(T object, Supplier<T> defaultSupplier) {
        assertEquals(ObjectUtils.getIfNull(object, defaultSupplier), object != null ? object : Suppliers.get(defaultSupplier));
    }

    private static Stream<Arguments> provideGetIfNullCases() {
        return Stream.of(
            of(1, () -> 2),
            of("a", () -> "b"),
            of(true, () -> false)

    @MethodSource("provideModeCases")
    public <T> void testMode(T... items) {
        assertEquals(ObjectUtils.mode(items), Arrays.stream(items).collect(Collectors.groupingBy(Function.identity(), Collectors.counting())).entrySet().stream()
