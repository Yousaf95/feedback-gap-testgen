package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import java.io.File;
import java.util.Properties;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.ValueSource;

public class SystemUtils_GenTest {

    private Properties systemProperties;

    @BeforeEach
    public void setUp() {
        systemProperties = new Properties();
        System.setProperties(systemProperties);
    }

    @AfterEach
    public void tearDown() {
        System.clearProperty("file.encoding");
        System.clearProperty("file.separator");
        System.clearProperty("java.awt.fonts");
        System.clearProperty("java.awt.graphicsenv");
        System.clearProperty("java.awt.headless");
        System.clearProperty("java.awt.printerjob");
        System.clearProperty("java.class.path");
        System.clearProperty("java.class.version");
        System.clearProperty("java.compiler");
        System.clearProperty("java.endorsed.dirs");
        System.clearProperty("java.ext.dirs");
        System.clearProperty("java.home");
        System.clearProperty("java.io.tmpdir");
        System.clearProperty("java.library.path");
        System.clearProperty("java.runtime.name");
        System.clearProperty("java.runtime.version");
        System.clearProperty("java.specification.name");
        System.clearProperty("java.specification.vendor");
        System.clearProperty("java.specification.version");
        System.clearProperty("java.util.prefs.PreferencesFactory");
        System.clearProperty("java.vendor");
        System.clearProperty("java.vendor.url");
        System.clearProperty("java.version");
        System.clearProperty("java.vm.info");
        System.clearProperty("java.vm.name");
        System.clearProperty("java.vm.specification.name");
        System.clearProperty("java.vm.specification.vendor");
        System.clearProperty("java.vm.specification.version");
        System.clearProperty("java.vm.vendor");
        System.clearProperty("java.vm.version");
        System.clearProperty("line.separator");
        System.clearProperty("os.arch");
        System.clearProperty("os.name");
        System.clearProperty("os.version");
        System.clearProperty("path.separator");
        System.clearProperty("user.country");
        System.clearProperty("user.dir");
        System.clearProperty("user.home");
        System.clearProperty("user.language");
        System.clearProperty("user.name");
        System.clearProperty("user.timezone");
    }

    @Test
    public void testGetFileEncoding() {
        systemProperties.setProperty("file.encoding", "UTF-8");
        assertEquals("UTF-8", SystemUtils.FILE_ENCODING);
    }

    @ParameterizedTest
    @ValueSource(strings = {"Windows 10", "Mac OS X", "Linux"})
    public void testIsOsMatchesName(String osName) {
        systemProperties.setProperty("os.name", osName);
        assertTrue(SystemUtils.getOsMatchesName(osName));
    }

    @Test
    public void testGetJavaHome() {
        systemProperties.setProperty("java.home", "/path/to/java/home");
        File javaHome = SystemUtils.getJavaHome();
        assertEquals("/path/to/java/home", javaHome.getPath());
    }

    @Test
    public void testIsJavaVersionAtLeast() {
        systemProperties.setProperty("java.specification.version", "17");
        assertTrue(SystemUtils.isJavaVersionAtLeast(JavaVersion.JAVA_17));
    }

    @Test
    public void testGetUserNameWithDefaultValue() {
        systemProperties.setProperty("user.name", "testUser");
        assertEquals("testUser", SystemUtils.getUserName("defaultUser"));
    }

    @Test
    public void testIsJavaAwtHeadless() {
        systemProperties.setProperty("java.awt.headless", "true");
        assertTrue(SystemUtils.isJavaAwtHeadless());
    }
}
