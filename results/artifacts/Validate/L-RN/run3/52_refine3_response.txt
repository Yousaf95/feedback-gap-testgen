package org.apache.commons.lang3;

import org.junit.jupiter.api.Test;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;

import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertEquals;

public class Validate_GenTest {

    @Test
    void testNoNullElementsCollectionWithNullElement() {
        Exception exception = assertThrows(IllegalArgumentException.class, () -> {
            Validate.noNullElements(Arrays.asList("a", null, "b"));
        });
        assertEquals("The validated collection contains null element at index: 1", exception.getMessage());
    }

    @Test
    void testNoNullElementsArrayWithNullElement() {
        Exception exception = assertThrows(IllegalArgumentException.class, () -> {
            Validate.noNullElements(new String[]{"a", null, "b"});
        });
        assertEquals("The validated array contains null element at index: 1", exception.getMessage());
    }

    @Test
    void testNotBlankWithWhitespaceString() {
        Exception exception = assertThrows(IllegalArgumentException.class, () -> {
            Validate.notBlank("   ");
        });
        assertEquals("The validated character sequence is blank", exception.getMessage());
    }

    @Test
    void testNotEmptyMapEmptyCase() {
        Exception exception = assertThrows(IllegalArgumentException.class, () -> {
            Validate.notEmpty(new HashMap<>());
        });
        assertEquals("The validated map is empty", exception.getMessage());
    }

    @Test
    void testValidIndexArrayNegativeIndex() {
        Exception exception = assertThrows(IndexOutOfBoundsException.class, () -> {
            Validate.validIndex(new String[]{"a", "b"}, -1);
        });
        assertEquals("The validated array index is invalid: -1", exception.getMessage());
    }

    @Test
    void testValidIndexCharSequenceNegativeIndex() {
        Exception exception = assertThrows(IndexOutOfBoundsException.class, () -> {
            Validate.validIndex("abc", -1);
        });
        assertEquals("The validated character sequence index is invalid: -1", exception.getMessage());
    }

    @Test
    void testIsTrueWithFalseExpressionAndCustomMessage() {
        Exception exception = assertThrows(IllegalArgumentException.class, () -> {
            Validate.isTrue(false, "Custom message for false expression");
        });
        assertEquals("Custom message for false expression", exception.getMessage());
    }
}
