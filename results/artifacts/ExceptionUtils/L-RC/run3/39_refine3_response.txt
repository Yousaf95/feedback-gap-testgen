package org.apache.commons.lang3.exception;

import org.junit.jupiter.api.Test;
import java.io.PrintStream;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.List;
import static org.junit.jupiter.api.Assertions.*;

public class ExceptionUtils_GenTest {

    @Test
    void testAsRuntimeException_withCheckedException() {
        Exception checkedException = new Exception("Checked Exception");
        RuntimeException runtimeException = assertThrows(RuntimeException.class, () -> ExceptionUtils.asRuntimeException(checkedException));
        assertEquals("Checked Exception", runtimeException.getMessage());
    }

    @Test
    void testGetCauseUsingMethodName_withNullMethodName() {
        Throwable throwable = new Exception();
        assertNull(ExceptionUtils.getCauseUsingMethodName(throwable, null));
    }

    @Test
    void testGetCauseUsingMethodName_withInvalidMethodName() {
        Throwable throwable = new Exception();
        assertNull(ExceptionUtils.getCauseUsingMethodName(throwable, "invalidMethod"));
    }

    @Test
    void testGetCauseUsingMethodName_withIllegalAccessException() throws NoSuchMethodException {
        Throwable throwable = new Exception();
        Method method = Exception.class.getDeclaredMethod("getMessage");
        method.setAccessible(false);
        ExceptionUtils.getCauseUsingMethodName(throwable, "getMessage"); // This should not throw an exception but log it internally
    }

    @Test
    void testGetCauseUsingMethodName_withInvocationTargetException() throws NoSuchMethodException {
        Throwable throwable = new Exception();
        Method method = Exception.class.getDeclaredMethod("getMessage");
        ExceptionUtils.getCauseUsingMethodName(throwable, "getMessage"); // This should not throw an exception but log it internally
    }

    @Test
    void testGetCauseUsingMethodName_withIllegalArgumentException() throws NoSuchMethodException {
        Throwable throwable = new Exception();
        Method method = Exception.class.getDeclaredMethod("getMessage");
        ExceptionUtils.getCauseUsingMethodName(throwable, "getMessage"); // This should not throw an exception but log it internally
    }

    @Test
    void testGetStackFrames_withNonNullThrowable() {
        Throwable throwable = new Exception("Test Exception");
        String[] stackFrames = ExceptionUtils.getStackFrames(throwable);
        assertTrue(stackFrames.length > 0);
    }

    @Test
    void testIndexOf_withNullThrowable() {
        assertEquals(ExceptionUtils.NOT_FOUND, ExceptionUtils.indexOf(null, Exception.class, 0, false));
    }

    @Test
    void testIndexOf_withNullType() {
        Throwable throwable = new Exception();
        assertEquals(ExceptionUtils.NOT_FOUND, ExceptionUtils.indexOf(throwable, null, 0, false));
    }

    @Test
    void testIndexOf_withNegativeFromIndex() {
        Throwable throwable = new Exception();
        int result = ExceptionUtils.indexOf(throwable, Exception.class, -1, false);
        assertEquals(0, result); // fromIndex should be adjusted to 0
    }

    @Test
    void testIndexOf_withLargeFromIndex() {
        Throwable throwable = new Exception();
        int result = ExceptionUtils.indexOf(throwable, Exception.class, Integer.MAX_VALUE, false);
        assertEquals(ExceptionUtils.NOT_FOUND, result); // fromIndex is larger than the number of throwables
    }

    @Test
    void testIndexOf_withSubclassTrue() {
        Throwable throwable = new RuntimeException("Test");
        int result = ExceptionUtils.indexOf(throwable, Exception.class, 0, true);
        assertEquals(0, result); // RuntimeException is a subclass of Exception
    }

    @Test
    void testIndexOf_withSubclassFalse() {
        Throwable throwable = new RuntimeException("Test");
        int result = ExceptionUtils.indexOf(throwable, Exception.class, 0, false);
        assertEquals(ExceptionUtils.NOT_FOUND, result); // RuntimeException is not exactly Exception
    }

    @Test
    void testHasCause_withUndeclaredThrowableException() {
        Throwable chain = new UndeclaredThrowableException(new NullPointerException("Root Cause"));
        assertTrue(ExceptionUtils.hasCause(chain, NullPointerException.class));
    }

    @Test
    void testPrintRootCauseStackTrace_withNonNullThrowableAndNullPrintStream() {
        Exception exception = new Exception();
        assertThrows(NullPointerException.class, () -> ExceptionUtils.printRootCauseStackTrace(exception, null));
    }

    @Test
    void testPrintRootCauseStackTrace_withNonNullThrowableAndNullPrintWriter() {
        Exception exception = new Exception();
        assertThrows(NullPointerException.class, () -> ExceptionUtils.printRootCauseStackTrace(exception, (PrintWriter) null));
    }

    @Test
    void testRemoveCommonFrames_withNonEmptyLists() {
        List<String> causeFrames = new ArrayList<>();
        causeFrames.add("Frame 1");
        causeFrames.add("Frame 2");

        List<String> wrapperFrames = new ArrayList<>();
        wrapperFrames.add("Frame 1");
        wrapperFrames.add("Frame 3");

        ExceptionUtils.removeCommonFrames(causeFrames, wrapperFrames);
        assertEquals(List.of("Frame 2"), causeFrames);
    }

    @Test
    void testRethrow_withCheckedException() {
        Exception checkedException = new Exception("Checked Exception");
        RuntimeException runtimeException = assertThrows(RuntimeException.class, () -> ExceptionUtils.rethrow(checkedException));
        assertEquals("Checked Exception", runtimeException.getMessage());
    }

    @Test
    void testThrowableOf_withNullThrowable() {
        assertNull(ExceptionUtils.throwableOf(null, Exception.class, 0, false));
    }

    @Test
    void testThrowableOf_withNullType() {
        Throwable throwable = new Exception();
        assertNull(ExceptionUtils.throwableOf(throwable, null, 0, false));
    }

    @Test
    void testThrowableOf_withNegativeFromIndex() {
        Throwable throwable = new Exception();
        int fromIndex = -1;
        ExceptionUtils.throwableOf(throwable, Exception.class, fromIndex, false); // This should not throw an exception but adjust fromIndex to 0 internally
    }

    @Test
    void testThrowableOf_withLargeFromIndex() {
        Throwable throwable = new Exception();
        int fromIndex = Integer.MAX_VALUE;
        assertNull(ExceptionUtils.throwableOf(throwable, Exception.class, fromIndex, false)); // fromIndex is larger than the number of throwables
    }

    @Test
    void testThrowableOf_withSubclassTrue() {
        Throwable throwable = new RuntimeException("Test");
        assertNotNull(ExceptionUtils.throwableOf(throwable, Exception.class, 0, true)); // RuntimeException is a subclass of Exception
    }

    @Test
    void testThrowableOf_withSubclassFalse() {
        Throwable throwable = new RuntimeException("Test");
        assertNull(ExceptionUtils.throwableOf(throwable, Exception.class, 0, false)); // RuntimeException is not exactly Exception
    }
}
