import org.junit.jupiter.api.Test;
import java.io.ByteArrayOutputStream;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import static org.junit.jupiter.api.Assertions.*;

class ExceptionUtils_GenTest {

    @Test
    public void testAsRuntimeException() {
        Throwable throwable = new RuntimeException("test");
        RuntimeException result = ExceptionUtils.asRuntimeException(throwable);
        assertEquals(RuntimeException.class, result.getClass());
        assertEquals("test", result.getMessage());
    }

    @Test
    public void testForEach() {
        Throwable throwable = new RuntimeException("test");
        List<String> messages = new ArrayList<>();
        ExceptionUtils.forEach(throwable, t -> messages.add(t.getMessage()));
        assertEquals(1, messages.size());
        assertEquals("test", messages.get(0));
    }

    @Test
    public void testGetCauseWithNullThrowable() {
        Throwable result = ExceptionUtils.getCause(null);
        assertNull(result);
    }

        Throwable result = ExceptionUtils.getCause(throwable, null);
        assertEquals(throwable, result);
    }

    @Test
    public void testGetCauseUsingMethodNameNoSuchMethodException() throws NoSuchMethodException {
        Throwable throwable = new RuntimeException("test");
        Method method = throwable.getClass().getMethod("nonExistentMethod");
        assertThrows(RuntimeException.class, () -> ExceptionUtils.getCauseUsingMethodName(throwable, "nonExistentMethod"));
    }

        ExceptionUtils.removeCommonFrames(causeFrames, wrapperFrames);
