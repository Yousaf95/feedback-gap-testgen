package org.apache.commons.lang3.exception;

import org.junit.jupiter.api.Test;

public class ExceptionUtils_GenTest {

    public void testGetCauseWithCustomMethodName() {
        Throwable cause = new RuntimeException("Root Cause");
        Throwable wrapper = new InvocationTargetException(cause);
        String[] methodNames = {"getTargetException"};
        Throwable result = ExceptionUtils.getCause(wrapper, methodNames);
        assertEquals(cause, result);
    }

    @Test
    public void testGetCauseWithNullMethodName() {

    }

        assertNull(result);
    }
