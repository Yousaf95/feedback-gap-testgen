package org.apache.commons.lang3.exception;

import java.io.PrintWriter;
import java.io.StringWriter;
import java.util.Arrays;
import java.util.List;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class ExceptionUtils_GenTest {

    @Test
    public void testAsRuntimeException() {
        Throwable throwable = new RuntimeException("test");
        RuntimeException result = ExceptionUtils.asRuntimeException(throwable);
        assertEquals(RuntimeException.class, result.getClass());
        assertEquals("test", result.getMessage());
    }

    @Test
    public void testForEach() {
        List<Throwable> throwables = Arrays.asList(new NullPointerException(), new IllegalArgumentException());
        ExceptionUtils.forEach(throwables.get(0), t -> assertTrue(throwables.contains(t)));
    }

    @Test
    public void testGetMessageWithNullThrowable() {
        String message = ExceptionUtils.getMessage(null);
        assertEquals("", message);
    }

    @Test
    public void testGetRootCauseMessageWithNullThrowable() {
        String rootCauseMessage = ExceptionUtils.getRootCauseMessage(null);
        assertEquals("", rootCauseMessage);
    }

    @Test
    public void testGetStackFramesWithNullThrowable() {
        String[] stackFrames = ExceptionUtils.getStackFrames(null);
        assertTrue(Arrays.equals(new String[0], stackFrames));
    }

    @Test
    public void testGetStackTraceWithNullThrowable() {
        String stackTrace = ExceptionUtils.getStackTrace(null);
        assertEquals("", stackTrace);
    }

    @Test
    public void testGetThrowablesWithNullThrowable() {
        Throwable[] throwables = ExceptionUtils.getThrowables(null);
        assertTrue(Arrays.equals(new Throwable[0], throwables));
    }

    @Test
    public void testHasCauseWithUndeclaredThrowableException() {
        Throwable cause = new UndeclaredThrowableException(new RuntimeException("test"));
        boolean result = ExceptionUtils.hasCause(cause, RuntimeException.class);
        assertTrue(result);
    }

    @Test
    public void testIndexOfThrowableWithNullThrowable() {
        int index = ExceptionUtils.indexOfThrowable(null, NullPointerException.class);
        assertEquals(-1, index);
    }

    @Test
    public void testIndexOfTypeWithNullThrowable() {
        int index = ExceptionUtils.indexOfType(null, NullPointerException.class);
        assertEquals(-1, index);
    }

    @Test
    public void testPrintRootCauseStackTraceWithNullThrowable() {
        assertDoesNotThrow(() -> ExceptionUtils.printRootCauseStackTrace(null));
    }

    @Test
    public void testPrintRootCauseStackTraceWithPrintStream() {
        String expected = "java.lang.RuntimeException: test\n";
        try (StringWriter sw = new StringWriter();
             PrintWriter pw = new PrintWriter(sw)) {
            ExceptionUtils.printRootCauseStackTrace(new RuntimeException("test"), pw);
            assertEquals(expected, sw.toString());
        }
    }

    @Test
    public void testPrintRootCauseStackTraceWithPrintWriter() {
        String expected = "java.lang.RuntimeException: test\n";
        try (StringWriter sw = new StringWriter();
             PrintWriter pw = new PrintWriter(sw)) {
            ExceptionUtils.printRootCauseStackTrace(new RuntimeException("test"), pw);
            assertEquals(expected, sw.toString());
        }
    }

    @Test
    public void testRemoveCommonFrames() {
        List<String> causeFrames = Arrays.asList(
                "com.example.CauseClass.method1",
                "com.example.WrapperClass.method2"
        );
        List<String> wrapperFrames = Arrays.asList(
                "com.example.WrapperClass.method2",
                "com.example.AnotherWrapperClass.method3"
        );
        ExceptionUtils.removeCommonFrames(causeFrames, wrapperFrames);
        assertEquals(Arrays.asList("com.example.CauseClass.method1"), causeFrames);
    }
}
