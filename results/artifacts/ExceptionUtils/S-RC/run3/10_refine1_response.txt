package org.apache.commons.lang3.exception;

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.util.Arrays;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.Stream;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class ExceptionUtils_GenTest {

    @Test
    public void testAsRuntimeException() {
        Throwable throwable = new RuntimeException("test");
        RuntimeException result = ExceptionUtils.asRuntimeException(throwable);
        assertEquals(RuntimeException.class, result.getClass());
        assertEquals("test", result.getMessage());
    }

    @Test
    public void testForEach() {
        List<Throwable> throwables = Arrays.asList(
            new IllegalArgumentException("Illegal"),
            new NullPointerException("Null")
        );
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        PrintStream printStream = new PrintStream(out);
        ExceptionUtils.forEach(throwables.get(0), t -> printStream.println(t));
        String result = out.toString();
        assertTrue(result.contains("java.lang.IllegalArgumentException: Illegal"));
    }

    @Test
    public void testGetCause() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        Throwable result = ExceptionUtils.getCause(throwable);
        assertEquals(IllegalArgumentException.class, result.getClass());
        assertEquals("Illegal", result.getMessage());
    }

    @Test
    public void testGetRootCause() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        Throwable rootCause = ExceptionUtils.getRootCause(throwable);
        assertEquals(IllegalArgumentException.class, rootCause.getClass());
        assertEquals("Illegal", rootCause.getMessage());
    }

    @Test
    public void testGetRootCauseMessage() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        String result = ExceptionUtils.getRootCauseMessage(throwable);
        assertTrue(result.contains("java.lang.IllegalArgumentException: Illegal"));
    }

    @Test
    public void testGetRootCauseStackTrace() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        List<String> stackTrace = ExceptionUtils.getRootCauseStackTraceList(throwable);
        assertTrue(stackTrace.stream().anyMatch(s -> s.contains("java.lang.IllegalArgumentException: Illegal")));
    }

    @Test
    public void testGetStackFrames() {
        Throwable throwable = new RuntimeException("Runtime");
        String[] stackFrames = ExceptionUtils.getStackFrames(throwable);
        assertTrue(Arrays.stream(stackFrames).anyMatch(s -> s.contains("ExceptionUtils_GenTest.testGetStackFrames")));
    }

    @Test
    public void testGetStackTrace() {
        Throwable throwable = new RuntimeException("Runtime");
        String stackTrace = ExceptionUtils.getStackTrace(throwable);
        assertTrue(stackTrace.contains("ExceptionUtils_GenTest.testGetStackTrace"));
    }

    @Test
    public void testGetThrowableCount() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        int count = ExceptionUtils.getThrowableCount(throwable);
        assertEquals(2, count);
    }

    @Test
    public void testHasCause() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        assertTrue(ExceptionUtils.hasCause(throwable, IllegalArgumentException.class));
    }

    @Test
    public void testIndexOfThrowable() {
        Throwable cause = new IllegalArgumentException("Illegal");
        Throwable throwable = new RuntimeException("Runtime", cause);
        int index = ExceptionUtils.indexOfThrowable(throwable, IllegalArgumentException.class);
        assertEquals(0, index);
    }

    @Test
    public void testIsChecked() {
        assertTrue(ExceptionUtils.isChecked(new IOException()));
        assertFalse(ExceptionUtils.isChecked(new RuntimeException()));
    }

    @Test
    public void testIsUnchecked() {
        assertFalse(ExceptionUtils.isUnchecked(new IOException()));
        assertTrue(ExceptionUtils.isUnchecked(new RuntimeException()));
    }

    @Test
    public void testPrintRootCauseStackTrace() {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        PrintStream printStream = new PrintStream(out);
        ExceptionUtils.printRootCauseStackTrace(new IllegalArgumentException("Illegal"), printStream);
        String result = out.toString();
        assertTrue(result.contains("java.lang.IllegalArgumentException: Illegal"));
    }
}
