package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Random;

public class RandomStringUtils_GenTest {

    @Test
    public void testRandomWithLargeCount() {
        String result = RandomStringUtils.random(10000);
        assertEquals(10000, result.length());
    }

    @Test
    public void testRandomAlphabeticWithLargeRange() {
        String result = RandomStringUtils.randomAlphabetic('A', 'Z');
        assertTrue(result.chars().allMatch(c -> c >= 'A' && c <= 'Z'));
    }

    @Test
    public void testRandomAlphanumericWithLargeRange() {
        String result = RandomStringUtils.randomAlphanumeric('0', '9');
        assertTrue(result.chars().allMatch(c -> c >= '0' && c <= '9'));
    }

    @Test
    public void testRandomAsciiWithLargeRange() {
        String result = RandomStringUtils.randomAscii(32, 127);
        assertTrue(result.chars().allMatch(c -> c >= 32 && c <= 127));
    }

    @Test
    public void testRandomGraphWithLargeRange() {
        String result = RandomStringUtils.randomGraph(33, 126);
        assertTrue(result.chars().allMatch(c -> c >= 33 && c <= 126));
    }

    @Test
    public void testRandomNumericWithLargeRange() {
        String result = RandomStringUtils.randomNumeric('0', '9');
        assertTrue(result.chars().allMatch(c -> c >= '0' && c <= '9'));
    }

    @Test
    public void testRandomPrintWithLargeRange() {
        String result = RandomStringUtils.randomPrint(32, 126);
        assertTrue(result.chars().allMatch(c -> c >= 32 && c <= 126));
    }

    @Test
    public void testRandomWithNonAsciiChars() {
        char[] chars = new char[10];
        for (int i = 0; i < chars.length; i++) {
            chars[i] = (char) (i + 128);
        }
        String result = RandomStringUtils.random(5, chars);
        assertTrue(result.chars().allMatch(c -> c >= 128));
    }

    @Test
    public void testRandomWithLargeNegativeCountThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.random(-1));
    }

    @Test
    public void testRandomAlphabeticWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomAlphabetic(-1));
    }

    @Test
    public void testRandomAlphanumericWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomAlphanumeric(-1));
    }

    @Test
    public void testRandomAsciiWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomAscii(-1));
    }

    @Test
    public void testRandomGraphWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomGraph(-1));
    }

    @Test
    public void testRandomNumericWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomNumeric(-1));
    }

    @Test
    public void testRandomPrintWithLargeNegativeLengthThrowsException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.randomPrint(-1));
    }
}
