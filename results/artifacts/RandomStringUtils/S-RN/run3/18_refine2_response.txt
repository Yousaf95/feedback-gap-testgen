package org.apache.commons.lang3;

import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.Test;
import java.util.Random;

public class RandomStringUtils_GenTest {

    @Test
    public void testRandomWithLargeCount() {
        String result = RandomStringUtils.random(10000);
        assertEquals(10000, result.length());
    }

    @Test
    public void testRandomAlphabeticRange_2() {
        String result = RandomStringUtils.randomAlphabetic(5, 'A', 'Z');
        assertTrue(result.matches("[A-Z]{5}"));
    }

    @Test
    public void testRandomAlphanumericRange_2() {
        String result = RandomStringUtils.randomAlphanumeric(5, '0', '9');
        assertTrue(result.matches("[0-9]{5}"));
    }

    @Test
    public void testRandomAsciiRange_2() {
        String result = RandomStringUtils.randomAscii(5, 33, 64);
        assertTrue(result.matches("[!@#$%^&*()]{5}"));
    }

    @Test
    public void testRandomGraphRange_2() {
        String result = RandomStringUtils.randomGraph(5, 33, 126);
        assertTrue(result.matches("[!\"#$%&'()*+,-./:;<=>?@\\[\\\\]^_`{|}~]{5}"));
    }

    @Test
    public void testRandomNumericRange_2() {
        String result = RandomStringUtils.randomNumeric(5, '0', '9');
        assertTrue(result.matches("[0-9]{5}"));
    }

    @Test
    public void testRandomPrintRange_2() {
        String result = RandomStringUtils.randomPrint(5, 32, 126);
        assertTrue(result.matches("[ !\"#$%&'()*+,-./:;<=>?@\\[\\\\]^_`{|}~]{5}"));
    }

    @Test
    public void testRandomWithCustomChars() {
        String result = RandomStringUtils.random(5, new char[]{'a', 'b', 'c'});
        assertTrue(result.matches("[abc]{5}"));
    }

    @Test
    public void testRandomWithEmptyCustomCharsThrowsIllegalArgumentException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.random(5, new char[0]));
    }
}
