package org.apache.commons.lang3;

import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

class RandomStringUtils_GenTest {

    @Test
    public void testRandomWithLargeCount() {
        String result = RandomStringUtils.random(10000);
        assertEquals(10000, result.length());
    }

    @Test
    public void testRandomAlphabeticRange_2_2() {
        String result = RandomStringUtils.randomAlphabetic(5);
        assertTrue(result.matches("[A-Za-z]{5}"));
    }

    @Test
    public void testRandomAlphanumericRange_2_2() {
        String result = RandomStringUtils.randomAlphanumeric(5);
        assertTrue(result.matches("[a-zA-Z0-9]{5}"));
    }

    @Test
    public void testRandomAsciiRange_2_2() {
        String result = RandomStringUtils.randomAscii(5);
        assertTrue(result.chars().allMatch(c -> c >= 33 && c <= 126));
    }

    @Test
    public void testRandomGraphRange_2_2() {
        String result = RandomStringUtils.randomGraph(5);
        assertTrue(result.chars().allMatch(c -> c >= 33 && c <= 126));
    }

    @Test
    public void testRandomNumericRange_2_2() {
        String result = RandomStringUtils.randomNumeric(5);
        assertTrue(result.matches("[0-9]{5}"));
    }

    @Test
    public void testRandomPrintRange_2_2() {
        String result = RandomStringUtils.randomPrint(5);
        assertTrue(result.chars().allMatch(c -> c >= 32 && c <= 126));
    }

    @Test
    public void testRandomWithCustomChars() {
        String result = RandomStringUtils.random(5, new char[]{'a', 'b', 'c'});
        assertTrue(result.matches("[abc]{5}"));
    }

    @Test
    public void testRandomWithEmptyCustomCharsThrowsIllegalArgumentException() {
        assertThrows(IllegalArgumentException.class, () -> RandomStringUtils.random(5, new char[0]));
    }
}
